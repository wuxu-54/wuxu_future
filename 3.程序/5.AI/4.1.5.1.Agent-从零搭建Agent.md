# 从零搭建一个生产级 Agent：规范实现与 Skill 集成完全指南

本文会带你从零开始，搭建一个完整的、可运行的、生产级 Agent 系统。  
它具备 **五要素**（感知、记忆、规划、工具使用、反思），采用 **ReAct** 架构，并支持 **动态发现与调用规范级 Skill**。  
读完本文，你将拥有一个可以直接用于学习、二次开发或作为项目基础模块的 Agent 实现。

---

## 一、项目概览

### 1.1 目标

构建一个 Agent，能够：

- 理解用户的自然语言指令
- 自主调用外部工具（包括本地函数和规范级 Skill）
- 记录短期记忆（对话上下文）和长期记忆（向量检索经验）
- 通过 ReAct 循环完成复杂任务
- 提供结构化的日志、错误处理和可观测性

### 1.2 技术栈

| 组件 | 技术选型 | 说明 |
|------|----------|------|
| 大模型 | OpenAI GPT-4o | 支持 function calling 和工具调用 |
| 编程语言 | Python 3.10+ | 主流 AI 工程语言 |
| 向量数据库 | ChromaDB | 轻量级长期记忆存储 |
| 配置管理 | python-dotenv + pydantic | 环境变量与配置校验 |
| 日志 | logging | 标准库，结构化输出 |
| 工具调用 | subprocess + OpenAI tool format | 支持规范级 Skill 的执行 |

```python
import logging

def setup_logger(name: str) -> logging.Logger:
    logger = logging.getLogger(name)
    if not logger.handlers:
        handler = logging.StreamHandler()
        formatter = logging.Formatter(
            "%(asctime)s - %(name)s - %(levelname)s - %(message)s"
        )
        handler.setFormatter(formatter)
        logger.addHandler(handler)
        logger.setLevel(logging.INFO)
    return logger
```

### 2.4 记忆模块 (`memory.py`)

实现短期记忆（滑动窗口消息列表）和长期记忆（ChromaDB 向量检索）。

```python
import chromadb
from chromadb.utils import embedding_functions
from typing import List, Dict, Any

class ShortTermMemory:
    """短期记忆：维护当前会话的消息列表"""
    def __init__(self, max_messages: int = 50):
        self.messages: List[Dict[str, str]] = []
        self.max_messages = max_messages

    def add(self, role: str, content: str):
        self.messages.append({"role": role, "content": content})
        if len(self.messages) > self.max_messages:
            self.messages.pop(0)  # 简单截断

    def get(self) -> List[Dict[str, str]]:
        return self.messages

    def clear(self):
        self.messages = []

class LongTermMemory:
    """长期记忆：使用 ChromaDB 存储嵌入向量"""
    def __init__(self, collection_name: str = "agent_memories"):
        self.client = chromadb.Client()
        self.embedding_fn = embedding_functions.DefaultEmbeddingFunction()
        self.collection = self.client.get_or_create_collection(
            name=collection_name,
            embedding_function=self.embedding_fn
        )

    def add(self, text: str, metadata: Dict[str, Any] = None):
        import hashlib
        doc_id = hashlib.md5(text.encode()).hexdigest()
        self.collection.upsert(
            documents=[text],
            metadatas=[metadata or {}],
            ids=[doc_id]
        )

    def search(self, query: str, top_k: int = 3) -> List[str]:
        results = self.collection.query(query_texts=[query], n_results=top_k)
        return results['documents'][0] if results['documents'] else []
```

### 2.5 工具与规范级 Skill 发现 (`skill_discovery.py`)

此模块负责扫描 `skills/` 目录，解析每个 Skill 的 `SKILL.md` 元数据，生成 OpenAI function calling 格式的工具定义，并提供执行函数。

```python
import os
import yaml
import subprocess
import json
from typing import Dict, Any, List

def discover_skills(skills_dir: str = "skills") -> List[Dict[str, Any]]:
    """返回 OpenAI 工具格式的列表，附带 _skill_path 内部字段"""
    tools = []
    if not os.path.exists(skills_dir):
        return tools

    for skill_name in os.listdir(skills_dir):
        skill_path = os.path.join(skills_dir, skill_name)
        skill_md = os.path.join(skill_path, "SKILL.md")
        if not os.path.isfile(skill_md):
            continue

        with open(skill_md, "r", encoding="utf-8") as f:
            content = f.read()

        # 解析 YAML frontmatter
        frontmatter = {}
        if content.startswith("---"):
            parts = content.split("---", 2)
            if len(parts) >= 2:
                try:
                    frontmatter = yaml.safe_load(parts[1])
                except yaml.YAMLError:
                    continue

        name = frontmatter.get("name", skill_name)
        description = frontmatter.get("description", "")
        inputs = frontmatter.get("inputs", [])

        properties = {}
        required = []
        for inp in inputs:
            prop = {
                "type": inp.get("type", "string"),
                "description": inp.get("description", "")
            }
            if "enum" in inp:
                prop["enum"] = inp["enum"]
            properties[inp["name"]] = prop
            if inp.get("required", False):
                required.append(inp["name"])

        tool_def = {
            "type": "function",
            "function": {
                "name": name,
                "description": description,
                "parameters": {
                    "type": "object",
                    "properties": properties,
                    "required": required
                }
            },
            "_skill_path": skill_path
        }
        tools.append(tool_def)
    return tools

def execute_skill(tool_name: str, arguments: Dict[str, Any], tools_meta: List[Dict]) -> str:
    """
    执行规范级 Skill：找到对应的脚本，运行并返回输出。
    约定：每个 skill 的入口脚本为 scripts/run.py，支持 --key value 参数。
    """
    for tool in tools_meta:
        if tool["function"]["name"] == tool_name:
            skill_path = tool["_skill_path"]
            script_path = os.path.join(skill_path, "scripts", "run.py")
            if not os.path.exists(script_path):
                return f"错误：Skill {tool_name} 未找到入口脚本 run.py"

            # 构建命令行参数
            cmd = ["python", script_path]
            for k, v in arguments.items():
                cmd.append(f"--{k}")
                cmd.append(str(v))
            try:
                result = subprocess.run(cmd, capture_output=True, text=True, timeout=30)
                if result.returncode != 0:
                    return f"Skill 执行失败 (code {result.returncode}): {result.stderr}"
                return result.stdout.strip()
            except subprocess.TimeoutExpired:
                return f"Skill {tool_name} 执行超时"
            except Exception as e:
                return f"执行 Skill 异常: {e}"
    return f"未知 Skill: {tool_name}"
```

### 2.6 可选的内置函数工具 (`tools.py`)

如果希望 Agent 也能使用本地函数（例如天气、计算器），可以定义普通函数工具，并合并到 Agent 的工具列表中。这里给出示例，但不强制。

```python
def get_weather(city: str) -> str:
    weather_db = {"北京": "晴天 25°C", "上海": "多云 22°C"}
    return weather_db.get(city, f"无法获取{city}天气")

def calculate(expression: str) -> str:
    try:
        safe_dict = {"__builtins__": {}}
        result = eval(expression, safe_dict, {})
        return f"计算结果: {result}"
    except Exception as e:
        return f"计算错误: {e}"

# 对应的 OpenAI tool 定义（可手动合并到 agent 的 tools 中）
BUILTIN_TOOLS = [
    {
        "type": "function",
        "function": {
            "name": "get_weather",
            "description": "获取城市天气",
            "parameters": {
                "type": "object",
                "properties": {"city": {"type": "string"}},
                "required": ["city"]
            }
        },
        "_impl": get_weather
    },
    {
        "type": "function",
        "function": {
            "name": "calculate",
            "description": "计算数学表达式",
            "parameters": {
```

### 2.7 Agent 核心实现 (`agent.py`)

整合以上所有模块，实现五要素和 ReAct 循环。

```python
import json
from openai import OpenAI
from config import Config
from logger_config import setup_logger
from memory import ShortTermMemory, LongTermMemory
from skill_discovery import discover_skills, execute_skill

logger = setup_logger("Agent")

class Agent:
    def __init__(self, skills_dir: str = "skills", use_builtin_tools: bool = True):
        self.client = OpenAI(api_key=Config.OPENAI_API_KEY)
        self.model = Config.OPENAI_MODEL
        self.max_steps = Config.MAX_STEPS

        # 记忆
        self.short_term = ShortTermMemory()
        self.long_term = LongTermMemory()

        # 加载规范级 Skill
        self.skill_tools = discover_skills(skills_dir)
        self.all_tools = self.skill_tools.copy()   # 先加入 Skill 工具

        if use_builtin_tools:
            from tools import BUILTIN_TOOLS
            self.all_tools.extend(BUILTIN_TOOLS)

        # 建立名称到执行函数的映射（对于内置工具）
        self.tool_impl_map = {}
        for t in self.all_tools:
            if "_impl" in t:
                self.tool_impl_map[t["function"]["name"]] = t["_impl"]

        logger.info(f"Agent 启动，共加载 {len(self.all_tools)} 个工具")

        # 系统提示
        tool_descs = "\n".join([f"- {t['function']['name']}: {t['function']['description']}" for t in self.all_tools])
        self.system_prompt = (
            "你是一个自主的智能体（Agent）。你可以使用以下工具来完成任务：\n"
            f"{tool_descs}\n\n"
            "请按照 ReAct 模式工作：\n"
            "1. 分析用户需求，决定下一步操作。\n"
            "2. 如果需要工具，调用合适的工具。\n"
            "3. 根据工具返回的结果，继续推理或给出最终答案。\n"
            "4. 当任务完成后，直接输出最终答案。"
        )
        self.short_term.add("system", self.system_prompt)

    def _build_messages(self, user_input: str) -> list:
        """构造发送给 LLM 的消息列表"""
        msgs = [{"role": "system", "content": self.system_prompt}]
        msgs.extend(self.short_term.get())
        msgs.append({"role": "user", "content": user_input})
        return msgs

    def _call_llm(self, messages: list):
        """调用 LLM，返回响应（原始对象）"""
        return self.client.chat.completions.create(
            model=self.model,
            messages=messages,
            tools=self.all_tools,
            tool_choice="auto",
            temperature=0
        )

    def run(self, user_input: str) -> str:
        logger.info(f"用户输入: {user_input}")

        # 1. 感知：获取用户输入
        # 2. 长期记忆检索（可选，增强上下文）
        memories = self.long_term.search(user_input, top_k=2)
        if memories:
            context = "相关历史经验：\n" + "\n".join(memories)
            self.short_term.add("system", context)

        messages = self._build_messages(user_input)
        step = 0

        while step < self.max_steps:
            step += 1
            logger.info(f"=== ReAct 步骤 {step} ===")

            try:
                response = self._call_llm(messages)
            except Exception as e:
                return f"Agent 调用 LLM 失败: {e}"

            message = response.choices[0].message
            # 记录到短期记忆
            self.short_term.add("assistant", message.content or "")
            messages.append(message)

            # 处理工具调用（包括规范级 Skill 和内置工具）
            if hasattr(message, 'tool_calls') and message.tool_calls:
                for tool_call in message.tool_calls:
                    tool_name = tool_call.function.name
                    arguments = json.loads(tool_call.function.arguments)

                    logger.info(f"调用工具: {tool_name}, 参数: {arguments}")

                    # 优先尝试内置工具，否则尝试 Skill 执行
                    if tool_name in self.tool_impl_map:
                        try:
                            result = self.tool_impl_map[tool_name](**arguments)
                            observation = str(result)
                        except Exception as e:
                            observation = f"内置工具执行错误: {e}"
                    else:
                        # 规范级 Skill 执行
                        observation = execute_skill(tool_name, arguments, self.skill_tools)

                    logger.info(f"工具返回: {observation}")

                    # 将工具结果作为 tool role 消息添加
                    tool_msg = {
                        "role": "tool",
                        "tool_call_id": tool_call.id,
                        "content": observation
                    }
                    messages.append(tool_msg)
                    self.short_term.add("tool", observation)
                # 本轮已处理工具，继续下一轮推理
                continue

            # 没有工具调用，则 message.content 应为最终答案
            if message.content:
                answer = message.content
                logger.info(f"最终答案: {answer}")
                # 3. 反思 & 长期记忆存储
                self.long_term.add(
                    text=f"用户问: {user_input}\nAgent 答: {answer}",
                    metadata={"steps": step, "success": True}
                )
                return answer

            # 既无工具调用也无内容，视为异常
            return "Agent 无法生成有效回答。"

        return "任务未完成，已达到最大执行步数。"

    def reset(self):
        """重置会话（清空短期记忆，保留长期记忆）"""
        self.short_term.clear()
        self.short_term.add("system", self.system_prompt)
```

### 2.8 运行入口 (`run.py`)

提供命令行交互界面。

```python
from agent import Agent

def main():
    agent = Agent(skills_dir="skills", use_builtin_tools=True)
    print("Agent 已启动。输入 'exit' 退出，输入 'reset' 重置会话。")
    while True:
        user_input = input("\n用户 > ").strip()
        if user_input.lower() == "exit":
            break
        if user_input.lower() == "reset":
            agent.reset()
            print("会话已重置。")
            continue
        if not user_input:
            continue
        response = agent.run(user_input)
        print(f"Agent > {response}")

if __name__ == "__main__":
    main()
```

## 五、生产级扩展建议

1. **使用更强大的长期记忆**  
   替换 ChromaDB 为 Pinecone、Weaviate 或 PostgreSQL+pgvector，支持更大规模。

2. **工具调用安全增强**  
   - 对内置工具（如 `calculate`）使用受限 `eval` 或 `ast.literal_eval`。  
   - 对规范级 Skill 执行设置超时、内存限制、沙箱（如 Docker 容器）。

3. **可观测性**  
   - 接入 OpenTelemetry 实现链路追踪。  
   - 记录每一步的 token 消耗和响应时间。

4. **异步与并发**  
   - 将 Agent 核心封装为异步函数（`async def`），支持多用户并发。  
   - 使用 Redis 存储短期记忆，支持水平扩展。

5. **Human-in-the-loop**  
   - 当 Agent 置信度低或遇到敏感操作时，要求用户确认。
