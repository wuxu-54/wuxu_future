## 1. 配置管理 (`src/config.py`)

```python
import os
from dotenv import load_dotenv

# 加载 .env 文件中的环境变量
load_dotenv()

class Settings:
    """应用配置类，从环境变量读取配置"""
    
    # 服务基本信息
    SERVER_NAME: str = os.getenv("MCP_SERVER_NAME", "Enterprise-MCP-Server")
    HOST: str = os.getenv("MCP_HOST", "0.0.0.0")          # 监听地址，0.0.0.0 表示允许外部访问
    PORT: int = int(os.getenv("MCP_PORT", "8000"))        # 服务端口
    
    # 安全配置：允许的 API Key 列表（逗号分隔）
    # 生产环境建议从数据库或密钥管理服务加载
    VALID_API_KEYS: set = set(os.getenv("MCP_API_KEYS", "").split(","))
    
    # 是否强制要求认证（为 True 时，所有请求必须携带有效的 API Key）
    AUTH_REQUIRED: bool = os.getenv("MCP_AUTH_REQUIRED", "True").lower() == "true"

# 创建全局配置实例，供其他模块导入
settings = Settings()
```

```python
"""
模拟企业级业务逻辑，实际使用时替换为真实的数据库/API 调用。
"""

# 模拟用户数据存储（字典）
user_db = {
    "12345": {"name": "张三", "email": "zhang.san@example.com", "department": "Engineering"},
    "67890": {"name": "李四", "email": "li.si@example.com", "department": "Sales"},
}

# 模拟订单数据存储（列表）
order_db = [
    {"id": "ORD-001", "user_id": "12345", "amount": 1250.00, "status": "delivered"},
    {"id": "ORD-002", "user_id": "12345", "amount": 89.90,   "status": "pending"},
    {"id": "ORD-003", "user_id": "67890", "amount": 3499.00, "status": "shipped"},
]


def get_user_profile(user_id: str) -> dict:
    """
    根据用户 ID 获取用户信息。
    
    参数:
        user_id: 用户唯一标识符
    
    返回:
        包含用户信息的字典，若用户不存在则返回错误信息。
    """
    user = user_db.get(user_id)
    if not user:
        return {"error": "User not found", "user_id": user_id}
    return user


def list_user_orders(user_id: str, limit: int = 5) -> list:
    """
    获取指定用户的订单列表。
    
    参数:
        user_id: 用户唯一标识符
        limit:   最多返回的订单数量
    
    返回:
        订单列表，每个订单是一个字典。
    """
    # 筛选属于该用户的订单
    user_orders = [o for o in order_db if o["user_id"] == user_id]
    # 按 limit 截取
    return user_orders[:limit]


def call_external_crm(user_id: str) -> dict:
    """
    模拟调用外部 CRM 系统，获取用户的信用评分等信息。
    实际生产环境中这里会发起 HTTP 请求到真实的 CRM API。
    
    参数:
        user_id: 用户唯一标识符
    
    返回:
        包含用户扩展信息的字典，若用户不存在则返回错误。
    """

    if "error" in user:
```

## 3. 服务核心逻辑 (`src/server.py`)

```python
import structlog
from fastmcp import FastMCP
from fastmcp.server.dependencies import get_http_headers
from . import tools
from .config import settings

# ===== 1. 配置结构化日志 =====
# structlog 生成 JSON 格式的日志，便于集中收集和分析
structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),   # 添加 ISO 格式时间戳
        structlog.processors.JSONRenderer()            # 输出为 JSON 字符串
    ]
)
logger = structlog.get_logger()   # 获取日志记录器实例

# ===== 2. 创建 MCP 服务器实例 =====
# FastMCP 是 MCP 协议的高级封装，负责处理 JSON-RPC 消息和工具注册
mcp = FastMCP(settings.SERVER_NAME)


# ===== 3. 认证中间件函数 =====
def verify_api_key():
    """
    从 HTTP 请求头中提取 Authorization: Bearer <token>，验证 API Key 是否合法。
    
    若认证失败，抛出 PermissionError；成功则返回 True。
    该函数会在每个需要认证的工具调用前被调用。
    """
    if not settings.AUTH_REQUIRED:
        return True
    
    # get_http_headers() 获取当前请求的 HTTP 头（FastMCP 提供的工具）
    headers = get_http_headers()
    auth_header = headers.get("authorization", "")
    
    # 移除 "Bearer " 前缀，得到纯 token
    api_key = auth_header.replace("Bearer ", "")
    
    # 检查 token 是否在配置的合法列表中
    if api_key not in settings.VALID_API_KEYS:
        logger.warning("auth_failed", api_key_prefix=api_key[:8] if api_key else "missing")
        raise PermissionError("Invalid or missing API Key")
    
    logger.info("auth_success", api_key_prefix=api_key[:8])
    return True


def log_tool_call(tool_name: str, args: dict, result: any, error: Exception = None):
    """
    统一的工具调用日志记录函数，记录每次工具调用的输入、输出和异常。
    
    参数:
        tool_name: 被调用的工具名称
        args:      调用时传入的参数
        result:    成功时返回的结果（可为 None）
        error:     失败时的异常对象（可为 None）
    """
    log_data = {
        "event": "tool_call",
        "tool_name": tool_name,
        "arguments": args,
    }
    if error:
        log_data.update({
            "status": "failed",
            "error_type": type(error).__name__,
            "error_message": str(error)
        })
    else:
        log_data["status"] = "success"
    # 发送结构化日志
    logger.info(**log_data)


# ===== 4. 注册 MCP 工具（每个工具都是一个可被 AI 调用的函数）=====

@mcp.tool
def query_user_profile(user_id: str) -> dict:
    """
    查询企业内部用户档案。
    
    参数:
        user_id (str): 用户的唯一标识符
    
    返回:
        dict: 包含用户姓名、邮箱、部门等信息。
    """
    # 1. 验证 API Key
    verify_api_key()
    
    try:
        # 2. 调用业务逻辑函数获取数据
        result = tools.get_user_profile(user_id)
        # 3. 记录成功日志
        log_tool_call("query_user_profile", {"user_id": user_id}, result)
        return result
    except Exception as e:
        # 4. 记录异常日志并返回友好的错误信息
        log_tool_call("query_user_profile", {"user_id": user_id}, None, e)
        return {"error": str(e)}


@mcp.tool
def get_user_orders(user_id: str, limit: int = 10) -> list:
    """
    获取指定用户的订单列表。
    
    参数:
        user_id (str): 用户的唯一标识符
        limit (int):   返回的最大订单数，范围 1-100（由 FastMCP 自动校验）
    
    返回:
        list: 订单对象列表，每个订单包含 id, user_id, amount, status。
    """
    verify_api_key()
    try:
        result = tools.list_user_orders(user_id, limit)
        log_tool_call("get_user_orders", {"user_id": user_id, "limit": limit}, result)
        return result
    except Exception as e:
        log_tool_call("get_user_orders", {"user_id": user_id, "limit": limit}, None, e)
        return {"error": str(e)}


@mcp.tool
def sync_crm_data(user_id: str) -> dict:
    """
    从外部 CRM 同步用户数据（敏感操作，需严格授权）。
    
    参数:
        user_id (str): 用户的唯一标识符
    
    返回:
        dict: 包含 CRM 信用评分、最近互动时间等信息。
    """
    verify_api_key()
    # 可在此处添加更细粒度的权限检查（例如：仅允许特定 API Key 调用）
    try:
        result = tools.call_external_crm(user_id)
        log_tool_call("sync_crm_data", {"user_id": user_id}, result)
        return result
    except Exception as e:
        log_tool_call("sync_crm_data", {"user_id": user_id}, None, e)
        return {"error": str(e)}
```

## 4. 服务入口 (`src/__main__.py`)

```python
from .server import mcp
from .config import settings

if __name__ == "__main__":
    # 打印启动信息

    print(f"Listening on http://{settings.HOST}:{settings.PORT}")
```

## 7. 通用同步客户端 (`client_sync.py`)

该客户端适用于任何编程语言，只需发送标准的 HTTP JSON-RPC 请求。

```python
import requests
import json

# ===== 配置 =====
MCP_URL = "http://localhost:8000/mcp"      # 服务端点
API_KEY = "admin-key-123"                   # 有效 API Key


def call_mcp_tool(tool_name: str, arguments: dict) -> dict:
    """
    调用 MCP 服务器上的工具。
    
    参数:
        tool_name:  工具名称（如 "query_user_profile"）
        arguments:  工具参数字典
    
    返回:
        服务器返回的 JSON-RPC 响应中的 result 部分。
    """
    # 构造 JSON-RPC 2.0 请求对象
    payload = {
        "jsonrpc": "2.0",
        "method": "tools/call",          # 固定方法名
        "params": {
            "name": tool_name,
            "arguments": arguments
        },
        "id": 1                           # 请求 ID，用于匹配响应
    }
    
    # 设置 HTTP 头
    headers = {
        "Content-Type": "application/json",
        "Authorization": f"Bearer {API_KEY}"   # 认证头
    }
    
    # 发送 POST 请求
    response = requests.post(MCP_URL, json=payload, headers=headers)
    
    # 解析 JSON 响应
    resp_json = response.json()
    
    # 检查是否存在错误
    if "error" in resp_json:
        raise Exception(f"RPC error: {resp_json['error']}")
    
    # 返回 result 部分
    return resp_json.get("result", {})


if __name__ == "__main__":
    # 示例 1: 查询用户信息
    print("=== 调用 query_user_profile ===")
    result = call_mcp_tool("query_user_profile", {"user_id": "12345"})
    print(json.dumps(result, indent=2, ensure_ascii=False))
    
    # 示例 2: 获取用户订单
    print("\n=== 调用 get_user_orders ===")
    result = call_mcp_tool("get_user_orders", {"user_id": "12345", "limit": 5})
    print(json.dumps(result, indent=2, ensure_ascii=False))
    
    # 示例 3: 同步 CRM 数据
    print("\n=== 调用 sync_crm_data ===")
    result = call_mcp_tool("sync_crm_data", {"user_id": "12345"})
    print(json.dumps(result, indent=2, ensure_ascii=False))
```

## 8. 异步智能体客户端（可选，`client_async.py`）

该示例集成 LangChain 和 Anthropic Claude，使 AI 能自动决策并调用工具。

```python
import asyncio
from langchain_mcp_adapters.client import MultiServerMCPClient
from langchain.agents import create_agent

async def run_ai_agent():
    """
    创建一个支持多 MCP 服务器的客户端，并让 AI 代理自动调用工具。
    """
    # 配置 MCP 服务器连接（支持同时连接多个）
    async with MultiServerMCPClient({
        "enterprise": {                         # 自定义服务器名称
            "url": "http://localhost:8000/mcp", # HTTP 端点
            "transport": "http",                # 使用 HTTP 传输
            "headers": {                        # 认证头
                "Authorization": "Bearer admin-key-123"
            }
        }
    }) as client:
        # 获取所有可用工具（自动发现）
        tools = await client.get_tools()
        
        # 创建 LangChain 代理，指定使用 Claude 3 模型（需设置 ANTHROPIC_API_KEY 环境变量）
        agent = create_agent(model="claude-3-opus-20240229", tools=tools)
        
        # 用户自然语言指令
        response = await agent.ainvoke({
            "messages": "请查询用户ID为12345的员工信息，并列出他的最近订单。"
        })
        
        # 打印最终回答
        print(response)

if __name__ == "__main__":
    asyncio.run(run_ai_agent())
```

> **注意**：运行异步客户端前需安装额外依赖：`pip install mcp anthropic langchain-mcp-adapters`，并设置 `ANTHROPIC_API_KEY` 环境变量。
