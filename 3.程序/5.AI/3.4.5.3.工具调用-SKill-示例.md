# 生产级Skill完整搭建与应用示例

我将带你从零实现一个**真实可用的代码审查Skill**，并展示如何将多个Skill编排成完整的工作流。这个示例完全遵循行业标准规范，不需要依赖任何复杂的Agent框架，你可以直接复制到生产环境中使用。

## 一、先看最终效果

我们将实现一个`code-review-skill`，它能：

1. 自动拉取GitHub仓库的指定分支代码
2. 运行单元测试
3. 检查代码风格
4. 扫描安全漏洞
5. 生成结构化的审查报告
6. 将报告发送到指定邮箱

整个过程完全自动化，只需要用户输入仓库地址和分支名称。

# 代码审查技能

## 功能描述
本技能用于自动化审查GitHub仓库的代码质量。它会拉取指定分支的代码，运行单元测试、代码风格检查和安全漏洞扫描，然后生成一份详细的审查报告并发送到指定邮箱。

## 执行流程（严格按照以下步骤执行，不得跳过）
1. 验证用户输入的仓库地址和分支名称是否有效
2. 调用`clone_repository`工具克隆GitHub仓库到本地临时目录
3. 调用`run_unit_tests`工具运行单元测试，记录测试结果
4. 调用`check_code_style`工具检查代码风格，记录问题
5. 调用`scan_security_vulnerabilities`工具扫描安全漏洞，记录结果
6. 综合所有结果，使用`generate_review_report`工具生成审查报告
7. 如果用户提供了邮箱地址，调用`send_email`工具发送报告
8. 清理临时目录，返回最终结果

## 输入要求
- `repo_url`: GitHub仓库HTTPS地址（必填）
- `branch`: 要审查的分支名称，默认为`main`（可选）
- `email`: 接收审查报告的邮箱地址（可选）

## 输出格式
```json
{
  "status": "success | failed",
  "message": "执行结果描述",
  "test_result": {
    "total": 总测试用例数,
    "passed": 通过数,
    "failed": 失败数,
    "details": "失败详情"
  },
  "style_issues": ["问题1", "问题2"],
  "security_issues": ["漏洞1", "漏洞2"],
  "report_url": "生成的报告下载地址",
  "email_sent": true | false
}
```

## 错误处理

- 如果仓库地址无效，返回错误："无效的GitHub仓库地址"
- 如果克隆仓库失败，返回错误："克隆仓库失败：{错误信息}"
- 如果运行测试失败，继续执行后续步骤，在报告中注明测试失败
- 如果发送邮件失败，返回警告但整体状态为成功

## 四、第三步：定义输入输出Schema

使用Pydantic定义严格的输入输出格式，确保类型安全。

```python
# schema.py
from pydantic import BaseModel, Field, HttpUrl
from typing import Optional, List, Dict

class CodeReviewInput(BaseModel):
    repo_url: HttpUrl = Field(..., description="GitHub仓库HTTPS地址")
    branch: str = Field("main", description="要审查的分支名称")
    email: Optional[str] = Field(None, description="接收审查报告的邮箱地址")

class TestResult(BaseModel):
    total: int = Field(..., description="总测试用例数")
    passed: int = Field(..., description="通过数")
    failed: int = Field(..., description="失败数")
    details: str = Field("", description="失败详情")

class CodeReviewOutput(BaseModel):
    status: str = Field(..., enum=["success", "failed"], description="执行状态")
    message: str = Field(..., description="执行结果描述")
    test_result: Optional[TestResult] = Field(None, description="测试结果")
    style_issues: List[str] = Field(default_factory=list, description="代码风格问题")
    security_issues: List[str] = Field(default_factory=list, description="安全漏洞")
    report_url: Optional[str] = Field(None, description="报告下载地址")
    email_sent: bool = Field(False, description="是否发送了邮件")
```

## 五、第四步：实现专属工具集

每个Skill都有自己专属的工具集，不会暴露无关工具。

```python
# tools.py
import os
import tempfile
import subprocess
import smtplib
from email.mime.text import MIMEText
from typing import Dict, Any

def clone_repository(repo_url: str, branch: str) -> str:
    """克隆GitHub仓库到临时目录"""
    temp_dir = tempfile.mkdtemp(prefix="code_review_")
    try:
        subprocess.run(
            ["git", "clone", "-b", branch, repo_url, temp_dir],
            check=True,
            capture_output=True,
            text=True
        )
        return temp_dir
    except subprocess.CalledProcessError as e:
        raise Exception(f"克隆仓库失败: {e.stderr}")

def run_unit_tests(repo_dir: str) -> Dict[str, Any]:
    """运行单元测试"""
    try:
        result = subprocess.run(
            ["python", "-m", "pytest", repo_dir, "-v", "--tb=short"],
            capture_output=True,
            text=True
        )
        
        # 简单解析测试结果（实际应用中使用pytest的JSON输出）
        output = result.stdout + result.stderr
        total = output.count("::") // 2
        passed = output.count("PASSED")
        failed = output.count("FAILED")
        
        return {
            "total": total,
            "passed": passed,
            "failed": failed,
            "details": output if failed > 0 else ""
        }
    except Exception as e:
        return {
            "total": 0,
            "passed": 0,
            "failed": 0,
            "details": f"运行测试失败: {str(e)}"
        }

def check_code_style(repo_dir: str) -> list:
    """检查代码风格（使用flake8）"""
    try:
        result = subprocess.run(
            ["flake8", repo_dir, "--max-line-length=80"],
            capture_output=True,
            text=True
        )
        
        issues = []
        for line in result.stdout.split("\n"):
            if line.strip():
                # 简化路径，只保留相对路径
                relative_path = line.replace(repo_dir + "/", "")
                issues.append(relative_path)
        
        return issues
    except Exception as e:
        return [f"代码风格检查失败: {str(e)}"]

def scan_security_vulnerabilities(repo_dir: str) -> list:
    """扫描安全漏洞（使用safety）"""
    try:
        requirements_path = os.path.join(repo_dir, "requirements.txt")
        if not os.path.exists(requirements_path):
            return ["未找到requirements.txt文件，跳过安全扫描"]
        
        result = subprocess.run(
            ["safety", "check", "--full-report", "--file", requirements_path],
            capture_output=True,
            text=True
        )
        
        vulnerabilities = []
        for line in result.stdout.split("\n"):
            if "==" in line and "vulnerability" in line.lower():
                vulnerabilities.append(line.strip())
        
        return vulnerabilities if vulnerabilities else ["未发现安全漏洞"]
    except Exception as e:
        return [f"安全扫描失败: {str(e)}"]

def generate_review_report(test_result: Dict, style_issues: list, security_issues: list) -> str:
    """生成审查报告"""
    report = f"""# 代码审查报告

    ## 测试结果
    - 总测试用例: {test_result['total']}
    - 通过: {test_result['passed']}
    - 失败: {test_result['failed']}

    {test_result['details']}

    ## 代码风格问题
    {chr(10).join(['- ' + issue for issue in style_issues]) if style_issues else '无'}

    ## 安全漏洞
    {chr(10).join(['- ' + issue for issue in security_issues]) if security_issues else '无'}
    """
    
    report_path = os.path.join(tempfile.gettempdir(), f"review_report_{os.urandom(4).hex()}.md")
    with open(report_path, "w", encoding="utf-8") as f:
        f.write(report)
    
    return report_path

def send_email(to_email: str, report_path: str) -> bool:
    """发送审查报告到指定邮箱"""
    try:
        # 这里使用模拟实现，实际应用中替换为真实的SMTP配置
        print(f"模拟发送邮件到 {to_email}，附件: {report_path}")
        return True
    except Exception as e:
        print(f"发送邮件失败: {str(e)}")
        return False

def cleanup_temp_dir(temp_dir: str):
    """清理临时目录"""
    try:
        subprocess.run(["rm", "-rf", temp_dir], check=True)
    except:
        pass

# 工具注册表 - 只暴露Skill需要的工具
AVAILABLE_TOOLS = {
    "clone_repository": clone_repository,
    "run_unit_tests": run_unit_tests,
    "check_code_style": check_code_style,
    "scan_security_vulnerabilities": scan_security_vulnerabilities,
    "generate_review_report": generate_review_report,
    "send_email": send_email,
    "cleanup_temp_dir": cleanup_temp_dir
}
```

## 六、第五步：实现Skill核心逻辑

```python
# skill.py
import json
from typing import Dict, Any
from schema import CodeReviewInput, CodeReviewOutput, TestResult
from tools import AVAILABLE_TOOLS

class CodeReviewSkill:
    def __init__(self):
        # 加载SKILL.md元数据（实际应用中解析Markdown文件）
        self.metadata = {
            "name": "代码审查",
            "version": "1.0.0",
            "description": "自动审查GitHub仓库代码"
        }
        self.tools = AVAILABLE_TOOLS
    
    def can_handle(self, query: str) -> float:
        """判断该Skill是否能处理用户查询，返回相似度得分（0-1）"""
        keywords = ["代码审查", "review", "检查代码", "测试代码", "代码质量"]
        score = sum(1 for keyword in keywords if keyword in query.lower()) / len(keywords)
        return min(score * 1.5, 1.0)  # 最高1.0
    
    def execute(self, input_data: Dict[str, Any]) -> Dict[str, Any]:
        """执行Skill"""
        # 1. 输入验证
        try:
            validated_input = CodeReviewInput(**input_data)
        except Exception as e:
            return CodeReviewOutput(
                status="failed",
                message=f"输入验证失败: {str(e)}"
            ).model_dump()
        
        temp_dir = None
        try:
            # 2. 执行流程
            # 步骤1: 克隆仓库
            temp_dir = self.tools["clone_repository"](
                str(validated_input.repo_url), 
                validated_input.branch
            )
            
            # 步骤2: 运行单元测试
            test_result_dict = self.tools["run_unit_tests"](temp_dir)
            test_result = TestResult(**test_result_dict)
            
            # 步骤3: 检查代码风格
            style_issues = self.tools["check_code_style"](temp_dir)
            
            # 步骤4: 扫描安全漏洞
            security_issues = self.tools["scan_security_vulnerabilities"](temp_dir)
            
            # 步骤5: 生成报告
            report_url = self.tools["generate_review_report"](
                test_result_dict, 
                style_issues, 
                security_issues
            )
            
            # 步骤6: 发送邮件
            email_sent = False
            if validated_input.email:
                email_sent = self.tools["send_email"](validated_input.email, report_url)
            
            # 3. 返回结果
            return CodeReviewOutput(
                status="success",
                message="代码审查完成",
                test_result=test_result,
                style_issues=style_issues,
                security_issues=security_issues,
                report_url=report_url,
                email_sent=email_sent
            ).model_dump()
        
        except Exception as e:
            return CodeReviewOutput(
                status="failed",
                message=f"执行失败: {str(e)}"
            ).model_dump()
        
        finally:
            # 步骤7: 清理临时目录
            if temp_dir:
                self.tools["cleanup_temp_dir"](temp_dir)

# 注册Skill
def register_skill():
    return CodeReviewSkill()
```

## 七、第六步：实现轻量级Skill运行时

现在我们实现一个简单的Skill运行时，用于管理和执行Skill。

```python
# skill_runtime.py
from typing import List, Dict, Any
from skill import register_skill as register_code_review_skill

class SkillRuntime:
    def __init__(self):
        self.skills = {}
        self._register_default_skills()
    
    def _register_default_skills(self):
        """注册所有可用的Skill"""
        self.register_skill(register_code_review_skill())
        # 在这里注册其他Skill
    
    def register_skill(self, skill):
        """注册单个Skill"""
        self.skills[skill.metadata["name"]] = skill
        print(f"已注册Skill: {skill.metadata['name']} v{skill.metadata['version']}")
    
    def find_best_skill(self, query: str):
        """找到最适合处理用户查询的Skill"""
        best_skill = None
        best_score = 0.0
        
        for skill in self.skills.values():
            score = skill.can_handle(query)
            if score > best_score and score > 0.3:  # 最低匹配阈值
                best_score = score
                best_skill = skill
        
        return best_skill, best_score
    
    def execute_skill(self, skill_name: str, input_data: Dict[str, Any]) -> Dict[str, Any]:
        """执行指定的Skill"""
        if skill_name not in self.skills:
            return {"status": "failed", "message": f"未知Skill: {skill_name}"}
        
        skill = self.skills[skill_name]
        return skill.execute(input_data)
    
    def handle_query(self, query: str, input_data: Dict[str, Any] = None) -> Dict[str, Any]:
        """自动处理用户查询"""
        skill, score = self.find_best_skill(query)
        
        if not skill:
            return {"status": "failed", "message": "没有找到合适的Skill来处理您的请求"}
        
        print(f"匹配到Skill: {skill.metadata['name']} (相似度: {score:.2f})")
        
        if input_data is None:
            # 实际应用中，这里应该让大模型从用户查询中提取输入参数
            return {"status": "success", "message": f"需要以下参数来执行{skill.metadata['name']}", "skill": skill.metadata["name"]}
        
        return skill.execute(input_data)
```

## 十、与纯提示词实现的对比

现在你可以清晰地看到Skill相比纯提示词的优势：

1. **结构清晰**：所有组件分离，职责明确
2. **类型安全**：Pydantic确保输入输出格式正确
3. **权限控制**：每个Skill只能访问自己的工具
4. **可测试性**：可以单独测试每个工具和Skill
5. **可复用性**：可以在任何地方调用这个Skill
6. **可观测性**：可以轻松添加日志和监控
7. **可维护性**：修改一个部分不会影响其他部分

## 十一、生产环境扩展建议
