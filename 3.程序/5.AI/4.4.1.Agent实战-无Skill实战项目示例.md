# Agent 规范级实现完整示例

本示例提供一个 **符合生产规范的 Agent 实现**，完全遵循五要素（感知、记忆、规划、工具使用、反思）和 ReAct 架构。代码不依赖 LangChain 等高级框架，仅使用 OpenAI SDK，便于理解和定制。包含完整的错误处理、日志、记忆管理、工具调用循环。

import logging

def setup_logger(name: str) -> logging.Logger:
    logger = logging.getLogger(name)
    if not logger.handlers:
        handler = logging.StreamHandler()
        formatter = logging.Formatter(
            "%(asctime)s - %(name)s - %(levelname)s - %(message)s"
        )
        handler.setFormatter(formatter)
        logger.addHandler(handler)
        logger.setLevel(logging.INFO)
    return logger

### 5. `memory.py` —— 记忆管理（短期 + 长期）

```python
import json
import chromadb
from chromadb.utils import embedding_functions
from typing import List, Dict, Any

class ShortTermMemory:
    """短期记忆：维护当前会话的消息历史"""
    def __init__(self, max_tokens: int = 4000):
        self.messages: List[Dict[str, str]] = []
        self.max_tokens = max_tokens  # 简单实现，未做实际 token 截断

    def add(self, role: str, content: str):
        self.messages.append({"role": role, "content": content})

    def get(self) -> List[Dict[str, str]]:
        return self.messages

    def clear(self):
        self.messages = []

class LongTermMemory:
    """长期记忆：使用 ChromaDB 存储嵌入向量，支持语义检索"""
    def __init__(self, collection_name: str = "agent_memories"):
        self.client = chromadb.Client()
        self.embedding_fn = embedding_functions.DefaultEmbeddingFunction()
        self.collection = self.client.get_or_create_collection(
            name=collection_name,
            embedding_function=self.embedding_fn
        )

    def add(self, text: str, metadata: Dict[str, Any] = None):
        """存储一段经验/知识"""
        self.collection.add(
            documents=[text],
            metadatas=[metadata or {}],
            ids=[f"mem_{hash(text)}"]
        )

    def search(self, query: str, top_k: int = 3) -> List[str]:
        """检索与查询最相关的记忆"""
        results = self.collection.query(query_texts=[query], n_results=top_k)
        return results['documents'][0] if results['documents'] else []
```

### 6. `tools.py` —— 工具定义

工具遵循 OpenAI function calling 格式。每个工具包含名称、描述、参数 schema 和实际执行函数。

```python
import json
from typing import Dict, Any

# 工具1：获取天气
def get_weather(city: str) -> str:
    """模拟天气 API"""
    # 实际可调用真实 API
    weather_db = {
        "北京": "晴天，25°C",
        "上海": "多云，22°C",
        "深圳": "阵雨，28°C"
    }
    return weather_db.get(city, f"无法获取{city}的天气")

# 工具2：计算器
def calculate(expression: str) -> str:
    """安全计算数学表达式"""
    try:
        # 仅允许数字和运算符，禁止危险函数
        if any(c in expression for c in ['__', 'import', 'os', 'sys']):
            return "错误：表达式包含非法字符"
        result = eval(expression, {"__builtins__": {}}, {})
        return f"计算结果：{result}"
    except Exception as e:
        return f"计算错误：{e}"

# 工具注册表（符合 OpenAI function calling 格式）
TOOLS = [
    {
        "type": "function",
        "function": {
            "name": "get_weather",
            "description": "获取指定城市的当前天气",
            "parameters": {
                "type": "object",
                "properties": {
                    "city": {"type": "string", "description": "城市名称，如'北京'"}
                },
                "required": ["city"]
            }
        }
    },
    {
        "type": "function",
        "function": {
            "name": "calculate",
            "description": "计算数学表达式，支持加减乘除和括号",
            "parameters": {
                "type": "object",
                "properties": {
                    "expression": {"type": "string", "description": "数学表达式，如'2+3*4'"}
                },
                "required": ["expression"]
            }
        }
    }
]

# 工具名到实际执行函数的映射
TOOL_IMPLEMENTATIONS = {
    "get_weather": get_weather,
    "calculate": calculate
}

def execute_tool(tool_name: str, arguments: Dict[str, Any]) -> str:
    """执行工具并返回结果字符串"""
    func = TOOL_IMPLEMENTATIONS.get(tool_name)
    if not func:
        return f"错误：未知工具 {tool_name}"
    try:
        return func(**arguments)
    except Exception as e:
        return f"工具执行失败：{e}"
```

### 7. `agent.py` —— Agent 核心实现（ReAct + 五要素）

```python
import json
from openai import OpenAI
from typing import Dict, Any, Optional
from config import Config
from logger_config import setup_logger
from memory import ShortTermMemory, LongTermMemory
from tools import TOOLS, execute_tool

logger = setup_logger("Agent")

class Agent:
    """
    符合五要素和 ReAct 架构的 Agent。
    包含：感知、记忆、规划、工具使用、反思。
    """
    def __init__(self):
        self.client = OpenAI(api_key=Config.OPENAI_API_KEY)
        self.model = Config.OPENAI_MODEL
        self.max_steps = Config.MAX_STEPS

        # 记忆组件
        self.short_term = ShortTermMemory()
        self.long_term = LongTermMemory()

        # 工具定义（OpenAI 格式）
        self.tools = TOOLS

        # 系统提示（指导 Agent 行为）
        self.system_prompt = (
            "你是一个自主的智能体（Agent）。你有以下能力：\n"
            "- 可以使用提供的工具来完成用户任务。\n"
            "- 你应该按照 Thought -> Action -> Observation 循环逐步推理。\n"
            "- 每一步，先思考（Thought）下一步要做什么，然后输出一个 Action。\n"
            "- Action 必须是以下格式：\n"
            "  Action: tool_name(arg1=value1, arg2=value2)\n"
            "  或者直接回答：Action: Final Answer: 你的回答\n"
            "- 当获得 Observation 后，继续下一轮 Thought。\n"
            "- 如果不确定，可以向用户提问。\n"
            "请遵循这个循环，直到给出最终答案。"
        )
        self.short_term.add("system", self.system_prompt)

    def _build_messages(self, user_input: str) -> list:
        """组装完整的消息列表：系统提示 + 短期记忆 + 当前输入"""
        messages = [{"role": "system", "content": self.system_prompt}]
        messages.extend(self.short_term.get())
        messages.append({"role": "user", "content": user_input})
        return messages

    def _call_llm(self, messages: list) -> Dict[str, Any]:
        """调用 LLM 并返回响应（自动处理工具调用）"""
        try:
            response = self.client.chat.completions.create(
                model=self.model,
                messages=messages,
                tools=self.tools,
                tool_choice="auto",
                temperature=0
            )
            return response.choices[0].message
        except Exception as e:
            logger.error(f"LLM 调用失败: {e}")
            raise

    def _parse_action(self, message) -> Optional[tuple]:
        """解析 LLM 返回的 Action（工具调用或最终答案）"""
        # 优先处理 tool_calls
        if hasattr(message, 'tool_calls') and message.tool_calls:
            tool_call = message.tool_calls[0]  # 简化：只处理第一个工具调用
            tool_name = tool_call.function.name
            arguments = json.loads(tool_call.function.arguments)
            return ("tool", tool_name, arguments, tool_call.id)

        # 处理文本中的 "Action: ..." 格式（兼容无 function calling 时）
        if message.content and "Action:" in message.content:
            lines = message.content.strip().split("\n")
            for line in lines:
                if line.startswith("Action:"):
                    action_str = line[7:].strip()
                    if action_str.startswith("Final Answer:"):
                        return ("final", action_str[13:].strip())
                    # 解析 tool_name(arg=val)
                    import re
                    match = re.match(r"(\w+)\((.*)\)", action_str)
                    if match:
                        tool_name = match.group(1)
                        args_str = match.group(2)
                        # 简单解析键值对
                        args = {}
                        for pair in args_str.split(","):
                            if "=" in pair:
                                k, v = pair.split("=", 1)
                                args[k.strip()] = v.strip().strip('"')
                        return ("tool", tool_name, args, None)
        # 如果直接返回最终答案
        if message.content:
            return ("final", message.content)
        return None

    def run(self, user_input: str) -> str:
        """Agent 主入口：执行 ReAct 循环"""
        logger.info(f"用户输入: {user_input}")

        # 1. 感知：接收用户输入
        # 2. 长期记忆检索（可选）
        memories = self.long_term.search(user_input, top_k=2)
        if memories:
            context = "相关历史经验：\n" + "\n".join(memories)
            self.short_term.add("system", context)  # 临时注入短期记忆

        # 初始化消息
        messages = self._build_messages(user_input)

        step = 0
        while step < self.max_steps:
            step += 1
            logger.info(f"步骤 {step}")

            # 3. 规划：调用 LLM 获取下一步动作
            try:
                message = self._call_llm(messages)
            except Exception as e:
                return f"Agent 内部错误：{e}"

            # 将 assistant 的消息加入短期记忆
            self.short_term.add("assistant", message.content or "")
            messages.append(message)

            # 解析动作
            action = self._parse_action(message)
            if not action:
                # 无有效动作，尝试直接返回内容
                return message.content or "抱歉，我无法处理这个请求。"

            action_type = action[0]

            # 4. 如果直接给出最终答案 -> 反思结束，返回
            if action_type == "final":
                answer = action[1]
                logger.info(f"最终答案: {answer}")
                # 5. 反思 + 长期记忆存储：将本次成功经验存入
                self.long_term.add(
                    text=f"用户问: {user_input} | 回答: {answer}",
                    metadata={"step": step, "success": True}
                )
                return answer

            # 处理工具调用
            if action_type == "tool":
                tool_name = action[1]
                args = action[2]
                tool_call_id = action[3]

                logger.info(f"调用工具: {tool_name}, 参数: {args}")

                # 4. 工具使用：执行工具并获得观察结果
                observation = execute_tool(tool_name, args)
                logger.info(f"观察结果: {observation}")

                # 将工具结果加入短期记忆
                tool_msg = {
                    "role": "tool",
                    "tool_call_id": tool_call_id,
                    "content": observation
                }
                messages.append(tool_msg)
                self.short_term.add("tool", observation)

                # 继续循环（反思将在下一轮的 Thought 中隐式进行）
                continue

            # 其他情况（如解析失败）
            return f"无法解析 Agent 动作: {action}"

        # 超过最大步数
        return "任务未完成，已达到最大执行步数。"

    def reset(self):
        """重置会话（清空短期记忆）"""
        self.short_term.clear()
        self.short_term.add("system", self.system_prompt)
```

### 8. `run.py` —— 启动示例

```python
from agent import Agent

def main():
    agent = Agent()
    print("Agent 已启动，输入 'exit' 退出。")
    while True:
        user_input = input("\n用户: ")
        if user_input.lower() == 'exit':
            break
        response = agent.run(user_input)
        print(f"Agent: {response}")
        # 可选：每次对话后不重置，保留上下文（短期记忆）
        # 如需新会话，调用 agent.reset()

if __name__ == "__main__":
    main()
```

## 四、规范说明

该实现严格遵循以下规范：

| 规范项 | 实现方式 |
|--------|----------|
| **五要素** | 感知（用户输入）、记忆（短期+长期）、规划（LLM 生成 Thought）、工具使用（execute_tool）、反思（下一次循环根据 Observation 调整） |
| **ReAct 架构** | Thought 隐含在 LLM 调用中，Action 通过 tool_calls 或文本解析，Observation 由工具返回并追加到消息历史 |
| **可观测性** | 结构化日志记录每一步 |
| **错误处理** | try-except 包裹 LLM 调用和工具执行 |
| **安全性** | 计算工具禁止危险字符，工具调用限制白名单 |
| **可扩展性** | 新增工具只需在 `tools.py` 中添加定义和实现 |

## 五、扩展建议

1. **强化规划**：可以显式要求 LLM 输出 `Thought:` 字段，并记录下来用于调试。
2. **多工具并行**：支持同时调用多个独立工具，提高效率。
3. **更丰富的记忆**：使用 ChromaDB 存储对话片段，并实现自动摘要压缩短期记忆。
4. **流式输出**：使用 `stream=True` 实现逐字响应。
5. **Human-in-the-loop**：当置信度低时，主动向用户提问澄清。

此示例提供了一个 **可直接运行、完全规范** 的 Agent 实现，是学习和生产定制的良好起点。
