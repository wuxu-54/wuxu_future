# 生产级RAG系统完整实现（LangChain+Milvus+BGE-M3）

这是一套**完全开源、可直接用于生产环境**的RAG系统实现，集成了所有核心优化技术：**递归语义分块、父文档检索、混合检索（向量+BM25）、BGE-M3重排序、上下文压缩**。全程不依赖OpenAI API，支持私有化部署。

### 2. 工具模块（utils.py）

数据清洗、日志等通用工具

```python
import re
from loguru import logger
import unicodedata

def clean_text(text: str) -> str:
    """生产级文本清洗函数"""
    # 统一换行符
    text = text.replace("\r\n", "\n").replace("\r", "\n")
    
    # 去除多余空白
    text = re.sub(r"\n{3,}", "\n\n", text)
    text = re.sub(r"[ \t]+", " ", text)
    
    # 去除不可见字符
    text = "".join(c for c in text if c.isprintable() or c in "\n\t")
    
    # 统一Unicode编码
    text = unicodedata.normalize("NFKC", text)
    
    # 去除首尾空白
    return text.strip()

def format_docs_with_sources(docs: list) -> str:
    """将检索到的文档格式化为带来源的上下文"""
    context = ""
    for i, doc in enumerate(docs, 1):
        source = doc.metadata.get("source", "未知来源")
        page = doc.metadata.get("page", "")
        page_str = f" 第{page}页" if page else ""
        
        context += f"【文档{i}】来源：{source}{page_str}\n"
        context += f"内容：{doc.page_content}\n\n"
    
    return context
```

### 3. 数据加载与分块模块（data_processor.py）

实现**递归语义分块 + 父文档检索**策略

```python
from langchain_community.document_loaders import (
    PyPDFLoader, Docx2txtLoader, TextLoader, UnstructuredMarkdownLoader
)
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain.storage import InMemoryStore
from langchain.retrievers import ParentDocumentRetriever
from utils import clean_text, logger
import os

class DataProcessor:
    def __init__(self):
        # 子文档分块器（用于检索）
        self.child_splitter = RecursiveCharacterTextSplitter(
            chunk_size=config.CHUNK_SIZE,
            chunk_overlap=config.CHUNK_OVERLAP,
            separators=["\n\n", "\n", ".", "。", "!", "！", "?", "？", " ", ""],
            is_separator_regex=False
        )
        
        # 父文档分块器（用于生成）
        self.parent_splitter = RecursiveCharacterTextSplitter(
            chunk_size=config.PARENT_CHUNK_SIZE,
            chunk_overlap=config.PARENT_CHUNK_OVERLAP
        )
        
        # 父文档存储
        self.docstore = InMemoryStore()
        
        # 支持的文件格式
        self.loader_map = {
            ".pdf": PyPDFLoader,
            ".docx": Docx2txtLoader,
            ".doc": Docx2txtLoader,
            ".txt": TextLoader,
            ".md": UnstructuredMarkdownLoader,
            ".markdown": UnstructuredMarkdownLoader
        }
    
    def load_single_document(self, file_path: str):
        """加载单个文档"""
        try:
            file_ext = os.path.splitext(file_path)[1].lower()
            loader_class = self.loader_map.get(file_ext)
            
            if not loader_class:
                logger.warning(f"不支持的文件格式：{file_ext}")
                return None
            
            loader = loader_class(file_path)
            docs = loader.load()
            
            # 清洗文本
            for doc in docs:
                doc.page_content = clean_text(doc.page_content)
                doc.metadata["source"] = os.path.basename(file_path)
            
            logger.info(f"成功加载文档：{file_path}，共{len(docs)}页")
            return docs
        
        except Exception as e:
            logger.error(f"加载文档失败：{file_path}，错误：{str(e)}")
            return None
    
    def load_documents_from_directory(self, directory: str):
        """加载目录下所有支持的文档"""
        all_docs = []
        for root, _, files in os.walk(directory):
            for file in files:
                file_path = os.path.join(root, file)
                docs = self.load_single_document(file_path)
                if docs:
                    all_docs.extend(docs)
        
        logger.info(f"共加载{len(all_docs)}个文档片段")
        return all_docs
    
    def get_parent_retriever(self, vectorstore):
        """创建父文档检索器"""
        return ParentDocumentRetriever(
            vectorstore=vectorstore,
            docstore=self.docstore,
            child_splitter=self.child_splitter,
            parent_splitter=self.parent_splitter
        )
```

### 4. 向量数据库模块（vector_db.py）

Milvus连接与索引配置

```python
from langchain_milvus import Milvus
from pymilvus import connections, Collection, FieldSchema, CollectionSchema, DataType
from utils import logger
import config

class VectorDB:
    def __init__(self, embedding_model):
        self.embedding_model = embedding_model
        self._connect()
    
    def _connect(self):
        """连接Milvus服务"""
        try:
            connections.connect(uri=config.MILVUS_URI)
            logger.info("成功连接Milvus服务")
        except Exception as e:
            logger.error(f"连接Milvus失败：{str(e)}")
            raise
    
    def get_vectorstore(self, collection_name: str = config.COLLECTION_NAME):
        """获取或创建向量存储"""
        # 定义索引参数（生产级HNSW配置）
        index_params = {
            "index_type": "HNSW",
            "metric_type": "COSINE",
            "params": {
                "M": 16,           # 每个节点的邻居数，平衡速度和召回率
                "ef_construction": 200  # 构建索引时的邻居探索数
            }
        }
        
        # 搜索参数
        search_params = {
            "metric_type": "COSINE",
            "params": {"ef": 128}  # 搜索时的邻居探索数
        }
        
        return Milvus(
            embedding_function=self.embedding_model,
            collection_name=collection_name,
            connection_args={"uri": config.MILVUS_URI},
            index_params=index_params,
            search_params=search_params,
            drop_old=False  # 生产环境设为False，避免误删数据
        )
    
    def delete_collection(self, collection_name: str = config.COLLECTION_NAME):
        """删除集合（谨慎使用）"""
        try:
            if connections.has_collection(collection_name):
                connections.drop_collection(collection_name)
                logger.info(f"成功删除集合：{collection_name}")
        except Exception as e:
            logger.error(f"删除集合失败：{str(e)}")
```

### 5. 检索引擎模块（retriever.py）

实现**混合检索 + BGE-M3重排序 + 上下文压缩**

```python
from langchain.retrievers import EnsembleRetriever, ContextualCompressionRetriever
from langchain_community.retrievers import BM25Retriever
from langchain.retrievers.document_compressors import CrossEncoderReranker
from utils import logger
import config

class RetrievalEngine:
    def __init__(self, vectorstore, all_docs):
        self.vectorstore = vectorstore
        self.all_docs = all_docs
        self._build_retrievers()
    
    def _build_retrievers(self):
        """构建多层检索引擎"""
        # 1. 向量检索器
        vector_retriever = self.vectorstore.as_retriever(
            search_kwargs={"k": config.TOP_K_RETRIEVE}
        )
        
        # 2. BM25关键词检索器
        bm25_retriever = BM25Retriever.from_documents(self.all_docs)
        bm25_retriever.k = config.TOP_K_RETRIEVE
        
        # 3. 混合检索器（Ensemble）
        self.ensemble_retriever = EnsembleRetriever(
            retrievers=[vector_retriever, bm25_retriever],
            weights=[config.HYBRID_WEIGHT_VECTOR, config.HYBRID_WEIGHT_BM25]
        )
        
        # 4. 重排序器（BGE-Reranker-v2-m3）
        logger.info("加载重排序模型...")
        compressor = CrossEncoderReranker(
            model_name=config.RERANKER_MODEL_NAME,
            top_n=config.TOP_K_RERANK,
            device="cuda" if torch.cuda.is_available() else "cpu"
        )
        
        # 5. 上下文压缩检索器
        self.compression_retriever = ContextualCompressionRetriever(
            base_compressor=compressor,
            base_retriever=self.ensemble_retriever
        )
        
        logger.info("检索引擎构建完成")
    
    def retrieve(self, query: str):
        """执行检索"""
        try:
            docs = self.compression_retriever.invoke(query)
            logger.info(f"检索完成，返回{len(docs)}个相关文档")
            return docs
        except Exception as e:
            logger.error(f"检索失败：{str(e)}")
            return []
```

### 6. 生成引擎模块（generator.py）

开源大模型生成与提示词工程

```python
from langchain_huggingface import HuggingFacePipeline
from transformers import AutoTokenizer, AutoModelForCausalLM, pipeline
from utils import format_docs_with_sources, logger
import config
import torch

class GenerationEngine:
    def __init__(self):
        self._load_model()
        self._build_prompt()
    
    def _load_model(self):
        """加载开源大模型"""
        logger.info(f"加载生成模型：{config.GENERATION_MODEL_NAME}...")
        
        self.tokenizer = AutoTokenizer.from_pretrained(config.GENERATION_MODEL_NAME)
        self.model = AutoModelForCausalLM.from_pretrained(
            config.GENERATION_MODEL_NAME,
            torch_dtype=torch.bfloat16,
            device_map="auto",
            trust_remote_code=True
        )
        
        # 创建生成管道
        pipe = pipeline(
            "text-generation",
            model=self.model,
            tokenizer=self.tokenizer,
            max_new_tokens=config.MAX_TOKENS,
            temperature=config.TEMPERATURE,
            top_p=0.9,
            repetition_penalty=1.05,
            do_sample=True
        )
        
        self.llm = HuggingFacePipeline(pipeline=pipe)
        logger.info("生成模型加载完成")
    
    def _build_prompt(self):
        """生产级RAG提示词"""
        self.system_prompt = """你是一个专业的知识助手，必须严格基于以下提供的参考信息回答用户的问题。

        规则：
        1. 只能使用参考信息中明确提到的内容，绝对不能编造信息
        2. 如果参考信息中没有答案，直接说"根据现有资料无法回答这个问题"
        3. 回答要准确、简洁、有条理
        4. 重要信息请标注来源，格式为【文档X】
        5. 不要提及"参考信息"、"根据文档"等类似表述，直接回答问题

        参考信息：
        {context}
        """
    
    def generate(self, query: str, docs: list):
        """生成回答"""
        try:
            if not docs:
                return "根据现有资料无法回答这个问题。"
            
            # 格式化上下文
            context = format_docs_with_sources(docs)
            
            # 构建提示词
            messages = [
                {"role": "system", "content": self.system_prompt.format(context=context)},
                {"role": "user", "content": query}
            ]
            
            # 生成回答
            response = self.llm.invoke(messages)
            return response
        
        except Exception as e:
            logger.error(f"生成回答失败：{str(e)}")
            return "抱歉，生成回答时发生错误。"
```

### 7. 主RAG流水线（rag_pipeline.py）

整合所有模块，提供统一接口

```python
from langchain_huggingface import HuggingFaceEmbeddings
from data_processor import DataProcessor
from vector_db import VectorDB
from retriever import RetrievalEngine
from generator import GenerationEngine
from utils import logger
import config
import torch

class RAGPipeline:
    def __init__(self):
        self.data_processor = DataProcessor()
        self._init_embedding_model()
        self.vector_db = VectorDB(self.embedding_model)
        self.generator = GenerationEngine()
        self.retrieval_engine = None
        self.vectorstore = None
    
    def _init_embedding_model(self):
        """初始化BGE-M3嵌入模型"""
        logger.info(f"加载嵌入模型：{config.EMBEDDING_MODEL_NAME}...")
        self.embedding_model = HuggingFaceEmbeddings(
            model_name=config.EMBEDDING_MODEL_NAME,
            model_kwargs={"device": "cuda" if torch.cuda.is_available() else "cpu"},
            encode_kwargs={"normalize_embeddings": True}
        )
        logger.info("嵌入模型加载完成")
    
    def build_knowledge_base(self, docs_directory: str):
        """构建知识库"""
        logger.info("开始构建知识库...")
        
        # 1. 加载所有文档
        all_docs = self.data_processor.load_documents_from_directory(docs_directory)
        if not all_docs:
            logger.error("没有加载到任何文档")
            return
        
        # 2. 获取向量存储
        self.vectorstore = self.vector_db.get_vectorstore()
        
        # 3. 创建父文档检索器并添加文档
        parent_retriever = self.data_processor.get_parent_retriever(self.vectorstore)
        parent_retriever.add_documents(all_docs)
        
        # 4. 构建检索引擎
        self.retrieval_engine = RetrievalEngine(self.vectorstore, all_docs)
        
        logger.info("知识库构建完成！")
    
    def query(self, question: str) -> dict:
        """查询接口"""
        if not self.retrieval_engine:
            raise ValueError("知识库尚未构建，请先调用build_knowledge_base()")
        
        logger.info(f"收到查询：{question}")
        
        # 1. 检索相关文档
        docs = self.retrieval_engine.retrieve(question)
        
        # 2. 生成回答
        answer = self.generator.generate(question, docs)
        
        # 3. 返回结果（包含回答和检索到的文档，便于调试）
        return {
            "question": question,
            "answer": answer,
            "retrieved_docs": [
                {
                    "content": doc.page_content,
                    "source": doc.metadata.get("source", "未知来源"),
                    "metadata": doc.metadata
                } for doc in docs
            ]
        }

# 主程序入口
if __name__ == "__main__":
    # 初始化RAG系统
    rag = RAGPipeline()
    
    # 构建知识库（将你的文档放在./docs目录下）
    rag.build_knowledge_base("./docs")
    
    # 交互式查询
    print("\n=== RAG系统已启动，输入'quit'退出 ===")
    while True:
        question = input("\n请输入你的问题：")
        if question.lower() in ["quit", "exit", "q"]:
            break
        
        result = rag.query(question)
        print("\n回答：")
        print(result["answer"])
```

## 四、生产级调优指南

### 1. 效果调优（按优先级排序）

| 问题 | 优化方向 | 具体操作 |
|------|----------|----------|
| 回答不准确/幻觉 | 提升检索质量 | 1. 增大`TOP_K_RETRIEVE`到30<br>2. 调整混合检索权重<br>3. 优化分块大小 |
| 检索不到相关内容 | 分块与检索策略 | 1. 减小`CHUNK_SIZE`到256<br>2. 启用语义分块<br>3. 使用查询改写 |
| 回答不完整 | 上下文完整性 | 1. 增大`PARENT_CHUNK_SIZE`到3072<br>2. 增大`TOP_K_RERANK`到8 |
| 回答包含无关信息 | 上下文过滤 | 1. 减小`TOP_K_RERANK`到3<br>2. 加强提示词约束 |

### 2. 性能调优

- **速度优化**：
  - 使用GPU加速（嵌入、重排序、生成都能显著提速）
  - 开启Milvus的索引缓存
  - 对热门查询结果进行缓存
- **内存优化**：
  - 使用量化模型（如4-bit/8-bit量化）
  - 减小批量大小
  - 对于超大规模知识库，使用Milvus分布式部署

### 3. 部署建议

- **模型选择**：
  - 嵌入：BGE-M3（综合最佳）
  - 重排序：BGE-Reranker-v2-m3（效果远超其他开源模型）
  - 生成：Qwen3-7B-Instruct（中文效果最好的7B模型）
- **生产环境**：
  - 使用Milvus分布式集群
  - 将模型部署为独立服务
  - 添加监控和日志系统
  - 实现知识库的增量更新
