# Function Call 实战笔记

mac可运行的部分轻量大模型不支持tools参数，执行会报错，所以有两种函数调用方式：1. 不支持tools工具的模型使用提示词进行调用  2. 支持tools工具的模型使用`tools`参数

## 方式一. 不支持tools工具的模型使用提示词进行调用

1. 定义工具函数
2. 定义工具描述tool结构

    ```py
        # --------------------------
        # 第二步：定义OpenAI SDK所需的tool格式配置
        # --------------------------
        tools = [
            {
                "type":"function",
                "function":{
                    'name':'calculator',
                    'description': "用于计算数学表达式的值",
                    "parameters":{
                        "type": "object",
                        "properties":{
                            "expression":{
                                "type": "string",
                                "description": "数学表达式"
                            }
                        },
                    }

                }
            },
            {
                "type":"function",
                "function":{
                    'name':'get_weather',
                    'description': "获取指定城市的天气",
                    "parameters":{
                        "type": "object",
                        "properties":{
                            "city":{
                                "type": "string",
                                "description": "城市名称"
                            },
                            "date":{
                                "type": "string",
                            }
                        },
                        "required": ["city"]
                    }
                }
            }
        ]

    ```

3. 创建提示词文本，将工具描述tool结构传入
4. 第一次大模型查询，配置message 内容使用："请分析问题，选择哪个工具函数进行调用，只返回工具函数具体名称，不写其他，如果无工具函数，请返回无"
5. 依据第一次查询返回的工具函数名称，调用对应的工具函数
6. 进行第二次大模型查询，message提示词内容修改："用简短的话说下这个结果{data}"  data为工具函数返回结果
7. 返回结果

```py
SYSTEM_PROMPT="""
    【系统】：
    你是一个智能助手，你需要根据用户问题，调用工具函数进行回答。
    """

def chat_with_function_call(self, question: str) -> str:
    USER_PROMPT=f"""
    【用户问题】：
    {question}
    【工具函数】：
    {self.tools}
    """

    message = [
                {
                    "role": "system",
                    "content": self.SYSTEM_PROMPT
                },
                    {
                    "role": "user",
                    "content": USER_PROMPT+"\n请分析问题，选择哪个工具函数进行调用，只返回工具函数具体名称，不写其他，如果无工具函数，请返回无"
                }
            ]


    try:
        response = self.client.chat.completions.create(
            model = "deepseek-r1:1.5b",
            messages=message,
        )

        print(response.choices[0].message.content)

        message_final = [
            {
                "role": "system",
                "content": self.SYSTEM_PROMPT
            }
        ]
        if "weather" in response.choices[0].message.content:
            weather = self.get_weather(city="北京",date="今天")
            message_final.append({"role":"user","content":f"用简短的话说下天气信息：{weather}"})
        elif "calculator" in response.choices[0].message.content:
            data = FunctionCall.calculator(expression="1+1")
            message_final.append({"role":"user","content":f"用简短的话说下这个结果：{data}"})

        # ---------- 再次调用模型，让模型基于工具结果生成最终回答 ----------
        final_response = self.client.chat.completions.create(
            model = "deepseek-r1:1.5b",
            messages=message_final
        )
        return final_response.choices[0].message.content
    except Exception as e:
        print(e)
        return "工具调用错误"
        
```

## 方式二. 支持tools工具的模型使用`tools`参数

大模型我选择的：Qwen2.5:7B

下载：ollama pull Qwen2.5:7B

1. 定义工具函数
2. 定义工具描述tool结构(同方式一)
3. 第一次大模型查询，大模型入参配置 tools, message正常配置，需要带入用户问题和工具函数
4. 遍历返回结果`response_message.tool_calls`，手动调用工具，并将结果添加到message内容，再次调用模型
6. 进行第二次大模型查询，返回结果

```py
SYSTEM_PROMPT="""
【系统】：
你是一个智能助手，你需要根据用户问题，调用工具函数进行回答。
"""
USER_PROMPT=f"""
【用户问题】：
{question}
【工具函数】：
{self.tools}
"""   

def chat_with_function_call(self, question: str) -> str:
    message = [
                {
                    "role": "system",
                    "content": self.SYSTEM_PROMPT
                },
                {
                    "role": "user",
                    "content": 
                        f"""
                        【用户问题】：
                        {question}

                        【工具函数】：
                        {self.tools}
                        """
                    
                }
            ]

    available_functions = {
        "calculator": self.calculator,
        "get_weather": self.get_weather,
    }


    try:
        response = self.client.chat.completions.create(
            model = "Qwen2.5:7b",
            messages=message,
            tools=self.tools,
            tool_choice="auto"
        )
        response_message = response.choices[0].message
        tool_calls = response_message.tool_calls
        if tool_calls:
            for tool_call in tool_calls:
                tool_id = tool_call.id
                function_name = tool_call.function.name
                function_args = json.loads(tool_call.function.arguments)
                print(f"调用工具函数：{function_name},args = {function_args}")

                function_to_call = available_functions.get(function_name)

                if function_to_call:
                    function_result = function_to_call(**function_args)  # 解包赋值
                else:
                    function_result = "工具函数不存在"
                
                message.append(
                    {
                        "role": "tool",
                        "content": function_result,
                        "tool_call_id": tool_id
                    }
                )
        # ---------- 再次调用模型，让模型基于工具结果生成最终回答 ----------
        final_response = self.client.chat.completions.create(
            model="Qwen2.5:7b",
            messages=message,
        )
        return final_response.choices[0].message.content
    except Exception as e:
        print(e)
        return "工具调用错误"
        
```
