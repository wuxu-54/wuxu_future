# Flutter 项目 AppTheme 实战配置与动态切换

## 一、项目结构

```
lib/
├── theme/
│   ├── app_colors.dart          # 颜色定义
│   ├── app_text_styles.dart     # 文字样式
│   ├── app_spacing.dart         # 间距/圆角/阴影
│   ├── app_theme.dart           # 主题配置入口
│   ├── theme_controller.dart    # 主题控制器（状态管理）
│   └── theme_preferences.dart   # 主题持久化
├── app.dart                     # MaterialApp 配置
└── main.dart
```

```
class AppRadius {
  static const double xs = 4.0;
  static const double sm = 6.0;
  static const double md = 8.0;
  static const double lg = 12.0;
  static const double xl = 16.0;
  static const double xxl = 24.0;
  static const double full = 999.0;
  
  // 组件圆角
  static const double card = 12.0;
  static const double button = 8.0;
  static const double input = 8.0;
  static const double sheet = 16.0;
  static const double dialog = 16.0;
}

class AppShadow {
  static List<BoxShadow> card(AppColors colors) => [
    BoxShadow(
      color: colors.shadow,
      blurRadius: 12,
      spreadRadius: 0,
      offset: const Offset(0, 2),
    ),
  ];
  
  static List<BoxShadow> button(AppColors colors) => [
    BoxShadow(
      color: colors.shadow,
      blurRadius: 6,
      spreadRadius: 0,
      offset: const Offset(0, 2),
    ),
  ];
  
  static List<BoxShadow> floating(AppColors colors) => [
    BoxShadow(
      color: colors.shadow,
      blurRadius: 16,
      spreadRadius: 0,
      offset: const Offset(0, 4),
    ),
  ];
}

class AppElevation {
  static const double card = 2.0;
  static const double button = 0.0;
  static const double appBar = 0.0;
  static const double bottomNav = 8.0;
  static const double fab = 4.0;
}
```

```
// lib/theme/app_theme.dart

class AppTheme {
  final AppColors colors;
  late final AppTextStyles textStyles;
  
  AppTheme(this.colors) {
    textStyles = AppTextStyles(colors);
  }
  
  ThemeData get themeData => ThemeData(
    useMaterial3: true,
    brightness: brightness,
    
    // 颜色方案
    colorScheme: colorScheme,
    
    // 脚手架背景
    scaffoldBackgroundColor: colors.background,
    
    // AppBar
    appBarTheme: _appBarTheme,
    
    // 文本主题
    textTheme: _textTheme,
    
    // 按钮主题
    elevatedButtonTheme: _elevatedButtonTheme,
    textButtonTheme: _textButtonTheme,
    outlinedButtonTheme: _outlinedButtonTheme,
    
    // 输入框
    inputDecorationTheme: _inputDecorationTheme,
    
    // 卡片
    cardTheme: _cardTheme,
    
    // 图标
    iconTheme: _iconTheme,
    
    // 分割线
    dividerTheme: _dividerTheme,
    
    // 底部导航
    bottomNavigationBarTheme: _bottomNavTheme,
    
    // TabBar
    tabBarTheme: _tabBarTheme,
    
    // 对话框
    dialogTheme: _dialogTheme,
    
    // BottomSheet
    bottomSheetTheme: _bottomSheetTheme,
    
    // 列表项
    listTileTheme: _listTileTheme,
    
    // 浮动按钮
    floatingActionButtonTheme: _fabTheme,
    
    // 开关
    switchTheme: _switchTheme,
    
    // 复选框
    checkboxTheme: _checkboxTheme,
    
    // 进度条
    progressIndicatorTheme: _progressTheme,
    
    // 滑块
    sliderTheme: _sliderTheme,
    
    // 导航栏（Android）
    navigationBarTheme: _navigationBarTheme,
    
    // 波纹颜色
    splashColor: colors.primary.withOpacity(0.1),
    highlightColor: colors.primary.withOpacity(0.05),
  );
  
  Brightness get brightness => 
      colors.background.computeLuminance() > 0.5 
          ? Brightness.light 
          : Brightness.dark;
  
  ColorScheme get colorScheme => ColorScheme(
    brightness: brightness,
    primary: colors.primary,
    onPrimary: colors.onPrimary,
    primaryContainer: colors.primaryContainer,
    onPrimaryContainer: colors.onPrimaryContainer,
    secondary: colors.secondary,
    onSecondary: colors.onSecondary,
    secondaryContainer: colors.secondaryContainer,
    onSecondaryContainer: colors.onSecondaryContainer,
    error: colors.error,
    onError: colors.onError,
    errorContainer: colors.errorContainer,
    onErrorContainer: colors.onErrorContainer,
    surface: colors.surface,
    onSurface: colors.onSurface,
    surfaceVariant: colors.surfaceVariant,
    onSurfaceVariant: colors.onSurfaceVariant,
    outline: colors.outline,
    outlineVariant: colors.outlineVariant,
    background: colors.background,
    onBackground: colors.onBackground,
    shadow: colors.shadow,
    scrim: colors.scrim,
    inverseSurface: colors.inverseSurface,
    onInverseSurface: colors.onInverseSurface,
    inversePrimary: colors.primary,
    tertiary: colors.info,
    onTertiary: colors.onInfo,
  );
  
  // ===== AppBar =====
  AppBarTheme get _appBarTheme => AppBarTheme(
    backgroundColor: colors.surface,
    foregroundColor: colors.textPrimary,
    elevation: 0,
    scrolledUnderElevation: 2,
    centerTitle: true,
    titleTextStyle: textStyles.h4,
    iconTheme: IconThemeData(color: colors.textPrimary, size: 24),
    actionsIconTheme: IconThemeData(color: colors.textPrimary, size: 24),
    systemOverlayStyle: SystemUiOverlayStyle(
      statusBarColor: Colors.transparent,
      statusBarIconBrightness: brightness == Brightness.light 
          ? Brightness.dark 
          : Brightness.light,
      statusBarBrightness: brightness,
    ),
  );
  
  // ===== 文本主题 =====
  TextTheme get _textTheme => TextTheme(
    displayLarge: textStyles.h1,
    displayMedium: textStyles.h2,
    displaySmall: textStyles.h3,
    headlineLarge: textStyles.h3,
    headlineMedium: textStyles.h4,
    headlineSmall: textStyles.h4,
    titleLarge: textStyles.h4,
    titleMedium: TextStyle(
      fontSize: 14,
      fontWeight: FontWeight.w500,
      color: colors.textPrimary,
    ),
    titleSmall: TextStyle(
      fontSize: 12,
      fontWeight: FontWeight.w500,
      color: colors.textSecondary,
    ),
    bodyLarge: textStyles.bodyLarge,
    bodyMedium: textStyles.bodyMedium,
    bodySmall: textStyles.bodySmall,
    labelLarge: textStyles.buttonMedium,
    labelMedium: TextStyle(
      fontSize: 12,
      fontWeight: FontWeight.w500,
      color: colors.textPrimary,
    ),
    labelSmall: textStyles.overline,
  );
  
  // ===== ElevatedButton =====
  ElevatedButtonThemeData get _elevatedButtonTheme => ElevatedButtonThemeData(
    style: ElevatedButton.styleFrom(
      backgroundColor: colors.primary,
      foregroundColor: colors.onPrimary,
      disabledBackgroundColor: colors.primary.withOpacity(0.4),
      disabledForegroundColor: colors.onPrimary.withOpacity(0.6),
      elevation: AppElevation.button,
      shadowColor: colors.shadow,
      padding: const EdgeInsets.symmetric(
        horizontal: AppSpacing.lg,
        vertical: AppSpacing.md,
      ),
      minimumSize: const Size(88, 44),
      shape: RoundedRectangleBorder(
        borderRadius: BorderRadius.circular(AppRadius.button),
      ),
      textStyle: textStyles.buttonMedium,
      tapTargetSize: MaterialTapTargetSize.shrinkWrap,
    ),
  );
  
  // ===== TextButton =====
  TextButtonThemeData get _textButtonTheme => TextButtonThemeData(
    style: TextButton.styleFrom(
      foregroundColor: colors.primary,
      padding: const EdgeInsets.symmetric(
        horizontal: AppSpacing.md,
        vertical: AppSpacing.sm,
      ),
      minimumSize: const Size(64, 36),
      shape: RoundedRectangleBorder(
        borderRadius: BorderRadius.circular(AppRadius.button),
      ),
      textStyle: textStyles.buttonMedium,
    ),
  );
  
  // ===== OutlinedButton =====
  OutlinedButtonThemeData get _outlinedButtonTheme => OutlinedButtonThemeData(
    style: OutlinedButton.styleFrom(
      foregroundColor: colors.primary,
      side: BorderSide(color: colors.primary, width: 1.5),
      padding: const EdgeInsets.symmetric(
        horizontal: AppSpacing.lg,
        vertical: AppSpacing.md,
      ),
      minimumSize: const Size(88, 44),
      shape: RoundedRectangleBorder(
        borderRadius: BorderRadius.circular(AppRadius.button),
      ),
      textStyle: textStyles.buttonMedium,
    ),
  );
  
  // ===== 输入框 =====
  InputDecorationTheme get _inputDecorationTheme => InputDecorationTheme(
    filled: true,
    fillColor: colors.surface,
    contentPadding: const EdgeInsets.symmetric(
      horizontal: AppSpacing.md,
      vertical: AppSpacing.md,
    ),
    hintStyle: textStyles.bodyMedium.copyWith(color: colors.textHint),
    labelStyle: TextStyle(color: colors.textSecondary),
    floatingLabelStyle: TextStyle(color: colors.primary),
    prefixIconColor: colors.textSecondary,
    suffixIconColor: colors.textSecondary,
    border: OutlineInputBorder(
      borderRadius: BorderRadius.circular(AppRadius.input),
      borderSide: BorderSide(color: colors.border),
    ),
    enabledBorder: OutlineInputBorder(
      borderRadius: BorderRadius.circular(AppRadius.input),
      borderSide: BorderSide(color: colors.border),
    ),
    focusedBorder: OutlineInputBorder(
      borderRadius: BorderRadius.circular(AppRadius.input),
      borderSide: BorderSide(color: colors.primary, width: 2),
    ),
    errorBorder: OutlineInputBorder(
      borderRadius: BorderRadius.circular(AppRadius.input),
      borderSide: BorderSide(color: colors.error),
    ),
    focusedErrorBorder: OutlineInputBorder(
      borderRadius: BorderRadius.circular(AppRadius.input),
      borderSide: BorderSide(color: colors.error, width: 2),
    ),
    disabledBorder: OutlineInputBorder(
      borderRadius: BorderRadius.circular(AppRadius.input),
      borderSide: BorderSide(color: colors.divider),
    ),
    errorStyle: TextStyle(color: colors.error, fontSize: 12),
  );
  
  // ===== 卡片 =====
  CardThemeData get _cardTheme => CardThemeData(
    color: colors.surface,
    elevation: AppElevation.card,
    shadowColor: colors.shadow,
    shape: RoundedRectangleBorder(
      borderRadius: BorderRadius.circular(AppRadius.card),
    ),
    margin: EdgeInsets.zero,
    clipBehavior: Clip.antiAlias,
  );
  
  // ===== 图标 =====
  IconThemeData get _iconTheme => IconThemeData(
    color: colors.textPrimary,
    size: 24,
  );
  
  // ===== 分割线 =====
  DividerThemeData get _dividerTheme => DividerThemeData(
    color: colors.divider,
    thickness: 1,
    space: 1,
  );
  
  // ===== 底部导航 =====
  BottomNavigationBarThemeData get _bottomNavTheme => BottomNavigationBarThemeData(
    backgroundColor: colors.surface,
    selectedItemColor: colors.primary,
    unselectedItemColor: colors.textSecondary,
    selectedLabelStyle: TextStyle(fontSize: 12, fontWeight: FontWeight.w500),
    unselectedLabelStyle: TextStyle(fontSize: 12),
    type: BottomNavigationBarType.fixed,
    elevation: AppElevation.bottomNav,
    showUnselectedLabels: true,
  );
  
  // ===== TabBar =====
  TabBarTheme get _tabBarTheme => TabBarTheme(
    labelColor: colors.primary,
    unselectedLabelColor: colors.textSecondary,
    labelStyle: TextStyle(fontSize: 14, fontWeight: FontWeight.w600),
    unselectedLabelStyle: TextStyle(fontSize: 14, fontWeight: FontWeight.normal),
    indicatorSize: TabBarIndicatorSize.label,
    indicatorColor: colors.primary,
    indicatorWeight: 2,
    dividerColor: colors.divider,
  );
  
  // ===== 对话框 =====
  DialogTheme get _dialogTheme => DialogTheme(
    backgroundColor: colors.surface,
    elevation: 8,
    shape: RoundedRectangleBorder(
      borderRadius: BorderRadius.circular(AppRadius.dialog),
    ),
    titleTextStyle: textStyles.h3,
    contentTextStyle: textStyles.bodyMedium,
  );
  
  // ===== BottomSheet =====
  BottomSheetThemeData get _bottomSheetTheme => BottomSheetThemeData(
    backgroundColor: colors.surface,
    elevation: 8,
    shape: RoundedRectangleBorder(
      borderRadius: BorderRadius.vertical(top: Radius.circular(AppRadius.sheet)),
    ),
    modalBackgroundColor: colors.surface,
    modalElevation: 8,
    clipBehavior: Clip.antiAlias,
  );
  
  // ===== ListTile =====
  ListTileThemeData get _listTileTheme => ListTileThemeData(
    iconColor: colors.textSecondary,
    textColor: colors.textPrimary,
    dense: false,
    contentPadding: EdgeInsets.symmetric(horizontal: AppSpacing.lg),
    shape: RoundedRectangleBorder(
      borderRadius: BorderRadius.circular(AppRadius.md),
    ),
  );
  
  // ===== FAB =====
  FloatingActionButtonThemeData get _fabTheme => FloatingActionButtonThemeData(
    backgroundColor: colors.primary,
    foregroundColor: colors.onPrimary,
    elevation: AppElevation.fab,
    shape: RoundedRectangleBorder(
      borderRadius: BorderRadius.circular(AppRadius.lg),
    ),
  );
  
  // ===== Switch =====
  SwitchThemeData get _switchTheme => SwitchThemeData(
    thumbColor: MaterialStateProperty.resolveWith((states) {
      if (states.contains(MaterialState.selected)) {
        return colors.primary;
      }
      return colors.outline;
    }),
    trackColor: MaterialStateProperty.resolveWith((states) {
      if (states.contains(MaterialState.selected)) {
        return colors.primary.withOpacity(0.5);
      }
      return colors.surfaceVariant;
    }),
  );
  
  // ===== Checkbox =====
  CheckboxThemeData get _checkboxTheme => CheckboxThemeData(
    fillColor: MaterialStateProperty.resolveWith((states) {
      if (states.contains(MaterialState.selected)) {
        return colors.primary;
      }
      return Colors.transparent;
    }),
    checkColor: MaterialStateProperty.all(colors.onPrimary),
    side: BorderSide(color: colors.outline, width: 2),
    shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(2)),
  );
  
  // ===== Progress =====
  ProgressIndicatorThemeData get _progressTheme => ProgressIndicatorThemeData(
    color: colors.primary,
    linearTrackColor: colors.surfaceVariant,
```

## 六、主题控制器（动态切换）

### 6.1 使用 Provider 实现

```
// lib/theme/theme_controller.dart
import 'package:provider/provider.dart';
import 'package:shared_preferences/shared_preferences.dart';

/// 主题模式
enum ThemeModeType {
  system,  // 跟随系统
  light,   // 亮色
  dark,    // 暗色
}

class ThemeController extends ChangeNotifier {
  static const String _key = 'theme_mode';
  
  ThemeModeType _mode = ThemeModeType.system;
  ThemeModeType get mode => _mode;
  
  // 获取当前主题
  AppTheme get currentTheme {
    switch (_mode) {
      case ThemeModeType.light:
        return lightTheme;
      case ThemeModeType.dark:
        return darkTheme;
      case ThemeModeType.system:
        // 跟随系统时，默认返回亮色，实际由 MaterialApp 的 themeMode 控制
        return lightTheme;
    }
  }
  
  // 获取 ThemeMode
  ThemeMode get themeMode {
    switch (_mode) {
      case ThemeModeType.light:
        return ThemeMode.light;
      case ThemeModeType.dark:
        return ThemeMode.dark;
      case ThemeModeType.system:
        return ThemeMode.system;
    }
  }
  
  // 初始化，从本地读取
  Future<void> init() async {
    final prefs = await SharedPreferences.getInstance();
    final index = prefs.getInt(_key);
    if (index != null && index < ThemeModeType.values.length) {
      _mode = ThemeModeType.values[index];
    }
    notifyListeners();
  }
  
  // 设置主题模式
  Future<void> setMode(ThemeModeType mode) async {
    if (_mode == mode) return;
    _mode = mode;
    notifyListeners();
    
    final prefs = await SharedPreferences.getInstance();
    await prefs.setInt(_key, mode.index);
  }
  
  // 切换亮/暗
  Future<void> toggle() async {
    if (_mode == ThemeModeType.dark) {
      await setMode(ThemeModeType.light);
    } else {
      await setMode(ThemeModeType.dark);
    }
  }
  
  // 是否是暗色模式
  bool isDark(BuildContext context) {
    if (_mode == ThemeModeType.dark) return true;
    if (_mode == ThemeModeType.light) return false;
    // 跟随系统
    return MediaQuery.of(context).platformBrightness == Brightness.dark;
  }
}
```

## 十二、总结

| 模块 | 职责 | 核心内容 |
| --- | --- | --- |
| `app_colors.dart` | 颜色定义 | 语义化颜色集，亮/暗两套 |
| `app_text_styles.dart` | 文字样式 | 标题/正文/按钮等层级 |
| `app_spacing.dart` | 间距系统 | 间距/圆角/阴影/高度 |
| `app_theme.dart` | 主题配置 | ThemeData + 所有组件主题 |
| `theme_controller.dart` | 状态管理 | 主题切换、持久化 |
| `theme_preferences.dart` | 持久化 | SharedPreferences 存储 |

**核心原则：**

1. **语义化命名**：用 `primary` / `success` 而非 `blue` / `red`
2. **单一数据源**：所有样式从主题获取，不硬编码
3. **亮暗双主题**：默认支持亮色和暗色两套
4. **持久化存储**：用户选择的主题保存到本地
5. **系统跟随**：支持跟随系统主题变化
6. **平滑过渡**：主题切换时有动画过渡
