# 构造函数

ArkTS 是基于 TypeScript 的，因此其构造函数机制与 TS/ES6 类似，但结合了鸿蒙声明式UI的特点。下面我将从基础到高级，全面解析构造函数的使用。

## 1. 构造函数基础

### 基本语法

```typescript
class Person {
  name: string;
  age: number;

  // 构造函数
  constructor(name: string, age: number) {
    this.name = name;
    this.age = age;
  }

  // 方法
  introduce(): string {
    return `我叫${this.name}，今年${this.age}岁`;
  }
}

// 使用
@Entry
@Component
struct Index {
  build() {
    let person = new Person("张三", 25);
    
    Column() {
      Text(person.introduce())
        .fontSize(20)
    }
  }
}
```

## 2. 在自定义组件中的构造函数

### 无参构造函数

```typescript
@Component
struct MyComponent {
  private count: number = 0;

  // 可选的构造函数
  constructor() {
    this.count = 10; // 初始化状态
  }

  build() {
    Column() {
      Text(`Count: ${this.count}`)
        .fontSize(20)
    }
  }
}
```

### 带参构造函数

```typescript
@Component
struct UserCard {
  private userName: string;
  private userAge: number;

  // 带参数的构造函数
  constructor(name: string, age: number) {
    this.userName = name;
    this.userAge = age;
  }

  build() {
    Column() {
      Text(`姓名: ${this.userName}`)
        .fontSize(18)
      Text(`年龄: ${this.userAge}`)
        .fontSize(16)
        .fontColor(Color.Gray)
    }
    .padding(10)
  }
}

@Entry
@Component
struct Index {
  build() {
    Column() {
      // 创建组件实例时传递参数
      UserCard({ name: "李四", age: 30 })
    }
  }
}
```

## 3. 构造函数的进阶用法

### 可选参数和默认值

```typescript
class Config {
  width: number;
  height: number;
  color: string;

  constructor(width: number = 100, height: number = 100, color?: string) {
    this.width = width;
    this.height = height;
    this.color = color || "#000000";
  }
}

@Component
struct ConfigurableBox {
  private config: Config;

  constructor(config?: Config) {
    this.config = config || new Config();
  }

  build() {
    Column() {
      Text(`宽度: ${this.config.width}`)
      Text(`高度: ${this.config.height}`)
      Text(`颜色: ${this.config.color}`)
    }
    .width(this.config.width)
    .height(this.config.height)
    .backgroundColor(this.config.color)
  }
}
```

### 参数属性简写

```typescript
class Point {
  // 直接在构造函数参数中声明和初始化属性
  constructor(public x: number = 0, public y: number = 0) {
    // 无需手动赋值，TypeScript/ArkTS 会自动处理
  }

  distance(): number {
    return Math.sqrt(this.x * this.x + this.y * this.y);
  }
}

@Component
struct PointDisplay {
  private point: Point = new Point(3, 4);

  build() {
    Column() {
      Text(`坐标: (${this.point.x}, ${this.point.y})`)
      Text(`距离: ${this.point.distance()}`)
    }
  }
}
```

## 4. 继承中的构造函数

### 父类构造函数调用

```typescript
// 基类
class Animal {
  name: string;

  constructor(name: string) {
    this.name = name;
  }

  speak(): string {
    return "动物叫声";
  }
}

// 派生类
class Dog extends Animal {
  breed: string;

  constructor(name: string, breed: string) {
    super(name); // 必须调用父类构造函数
    this.breed = breed;
  }

  // 重写方法
  speak(): string {
    return "汪汪汪";
  }
}

@Component
struct AnimalDemo {
  build() {
    let dog = new Dog("旺财", "金毛");
    
    Column() {
      Text(`名字: ${dog.name}`)
      Text(`品种: ${dog.breed}`)
      Text(`叫声: ${dog.speak()}`)
    }
  }
}
```

## 5. 私有构造函数与单例模式

```typescript
class AppSettings {
  private static instance: AppSettings;
  private theme: string = "light";
  private language: string = "zh-CN";

  // 私有构造函数防止外部实例化
  private constructor() {}

  // 静态方法获取单例
  public static getInstance(): AppSettings {
    if (!AppSettings.instance) {
      AppSettings.instance = new AppSettings();
    }
    return AppSettings.instance;
  }

  public setTheme(theme: string): void {
    this.theme = theme;
  }

  public getTheme(): string {
    return this.theme;
  }
}

@Component
struct SettingsDisplay {
  private settings: AppSettings = AppSettings.getInstance();

  build() {
    Column() {
      Text(`当前主题: ${this.settings.getTheme()}`)
      
      Button("切换主题")
        .onClick(() => {
          this.settings.setTheme(
            this.settings.getTheme() === "light" ? "dark" : "light"
          );
        })
    }
  }
}
```

## 6. 在 @Entry 和 @Component 中的实践

### 完整的组件示例

```typescript
// 数据模型
class Product {
  constructor(
    public id: number,
    public name: string,
    public price: number,
    public category: string = "默认分类"
  ) {}
}

@Component
struct ProductItem {
  private product: Product;

  constructor(product: Product) {
    this.product = product;
  }

  build() {
    Row() {
      Column() {
        Text(this.product.name)
          .fontSize(18)
          .fontWeight(FontWeight.Bold)
        Text(`¥${this.product.price}`)
          .fontSize(16)
          .fontColor(Color.Red)
        Text(this.product.category)
          .fontSize(12)
          .fontColor(Color.Gray)
      }
      .alignItems(HorizontalAlign.Start)
    }
    .width('100%')
    .padding(10)
    .backgroundColor(Color.White)
    .borderRadius(8)
    .margin({ bottom: 10 })
  }
}

@Entry
@Component
struct ProductList {
  private products: Product[] = [
    new Product(1, "华为手机", 5999, "电子产品"),
    new Product(2, "鸿蒙开发板", 299, "开发工具"),
    new Product(3, "智能手表", 1299)
  ];

  build() {
    Column() {
      Text("商品列表")
        .fontSize(24)
        .fontWeight(FontWeight.Bold)
        .margin({ bottom: 20 })

      // 使用构造函数创建组件实例
      ForEach(this.products, (item: Product) => {
        ProductItem({ product: item })
      }, (item: Product) => item.id.toString())
    }
    .padding(20)
    .width('100%')
    .height('100%')
    .backgroundColor('#F1F3F5')
  }
}
```

### 2. **使用 Builder 模式处理复杂构造**

```typescript
class HttpRequest {
  private method: string;
  private url: string;
  private headers: Map<string, string>;
  private body: string;

  private constructor(builder: HttpRequestBuilder) {
    this.method = builder.method;
    this.url = builder.url;
    this.headers = builder.headers;
    this.body = builder.body;
  }

  static Builder = class {
    method: string = "GET";
    url: string = "";
    headers: Map<string, string> = new Map();
    body: string = "";

    setMethod(method: string): this {
      this.method = method;
      return this;
    }

    setUrl(url: string): this {
      this.url = url;
      return this;
    }

    build(): HttpRequest {
      return new HttpRequest(this);
    }
  }
}

// 使用
let request = new HttpRequest.Builder()
  .setMethod("POST")
  .setUrl("https://api.example.com")
  .build();
```

## 关键要点总结

1. **必须性**: 在类中，构造函数是可选的，如果不定义，会有默认的无参构造函数

2. **继承要求**: 派生类的构造函数必须调用 `super()`，且必须在访问 `this` 之前调用

3. **组件使用**: 在自定义组件中，构造函数常用于初始化状态和接收参数

4. **访问控制**: 使用 `private constructor` 实现单例模式

5. **参数属性**: 使用参数属性可以简化代码，直接在构造函数中声明和初始化属性

6. **鸿蒙特色**: 在 ArkTS 中，构造函数与声明式UI紧密结合，支持创建可复用的组件实例

掌握好构造函数的使用，能够帮助你构建更加健壮和可维护的鸿蒙应用。
