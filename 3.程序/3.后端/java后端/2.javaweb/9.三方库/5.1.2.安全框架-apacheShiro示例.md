# Jeecg Boot 集成 Shiro + JWT + 动态权限 扩展配置模板

该模板完全适配 Jeecg Boot 项目结构，实现 **JWT 无状态认证** + **数据库动态权限** + **全局用户信息获取**，可直接复制到项目中使用。

## 一、核心依赖（确认 pom.xml 已引入）

Jeecg Boot 项目默认已集成 Shiro，只需补充 JWT 依赖：

```xml

```

## 二、JWT 工具类（生成/解析令牌）

```java
package com.jeecg.common.util;

import io.jsonwebtoken.*;
import io.jsonwebtoken.security.Keys;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import javax.crypto.SecretKey;
import java.util.Date;

/**
 * JWT 工具类（Jeecg Boot 适配版）
 */
@Slf4j
@Component
public class JwtUtil {

    // JWT 密钥（配置在 application.yml 中）
    @Value("${jeecg.shiro.jwt.secret:jeecg-boot-secret-key-1234567890}")
    private String secret;

    // 令牌过期时间：2小时（配置在 application.yml 中）
    @Value("${jeecg.shiro.jwt.expire:7200000}")
    private Long expire;

    // 生成签名密钥
    private SecretKey getSecretKey() {
        return Keys.hmacShaKeyFor(secret.getBytes());
    }

    /**
     * 生成 JWT 令牌（传入用户名）
     */
    public String generateToken(String username) {
        return Jwts.builder()
                .setSubject(username) // 主题：存储用户名
                .setIssuedAt(new Date()) // 签发时间
                .setExpiration(new Date(System.currentTimeMillis() + expire)) // 过期时间
                .signWith(getSecretKey(), SignatureAlgorithm.HS256) // 签名算法
                .compact();
    }

    /**
     * 解析令牌，获取用户名
     */
    public String getUsernameFromToken(String token) {
        try {
            Claims claims = Jwts.parserBuilder()
                    .setSigningKey(getSecretKey())
                    .build()
                    .parseClaimsJws(token)
                    .getBody();
            return claims.getSubject();
        } catch (ExpiredJwtException e) {
            log.error("JWT 令牌已过期", e);
            throw new RuntimeException("令牌已过期");
        } catch (Exception e) {
            log.error("JWT 令牌解析失败", e);
            throw new RuntimeException("令牌无效");
        }
    }

    /**
     * 验证令牌有效性
     */
    public boolean validateToken(String token) {
        try {
            Jwts.parserBuilder().setSigningKey(getSecretKey()).build().parseClaimsJws(token);
            return true;
        } catch (Exception e) {
            return false;
        }
    }
}
```

## 三、自定义 Shiro Realm（认证+授权）

适配 Jeecg Boot 的 `SysUser`、`SysRole`、`SysPermission` 表结构：

```java
package com.jeecg.common.shiro;

import com.jeecg.modules.system.entity.SysUser;
import com.jeecg.modules.system.service.ISysUserService;
import org.apache.shiro.authc.*;
import org.apache.shiro.authz.AuthorizationInfo;
import org.apache.shiro.authz.SimpleAuthorizationInfo;
import org.apache.shiro.realm.AuthorizingRealm;
import org.apache.shiro.subject.PrincipalCollection;
import org.apache.shiro.util.ByteSource;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

import java.util.List;

/**
 * Jeecg Boot 自定义 Shiro Realm
 */
@Component
public class JeecgShiroRealm extends AuthorizingRealm {

    @Autowired
    private ISysUserService sysUserService;

    // 必须重写此方法，否则 Shiro 无法识别 JWT Token
    @Override
    public boolean supports(AuthenticationToken token) {
        return token instanceof JwtToken || token instanceof UsernamePasswordToken;
    }

    /**
     * 授权：从数据库查询用户的角色和权限
     */
    @Override
    protected AuthorizationInfo doGetAuthorizationInfo(PrincipalCollection principals) {
        // 1. 获取当前登录用户（认证时存入的 SysUser 对象）
        SysUser user = (SysUser) principals.getPrimaryPrincipal();

        // 2. 构建授权信息
        SimpleAuthorizationInfo info = new SimpleAuthorizationInfo();

        // 3. 从数据库查询用户角色（Jeecg 自带方法）
        List<String> roleCodes = sysUserService.queryUserRoleCodes(user.getUsername());
        info.addRoles(roleCodes);

        // 4. 从数据库查询用户权限（Jeecg 自带方法）
        List<String> permissions = sysUserService.queryUserPermissions(user.getUsername());
        info.addStringPermissions(permissions);

        return info;
    }

    /**
     * 认证：验证用户名和密码
     */
    @Override
    protected AuthenticationInfo doGetAuthenticationInfo(AuthenticationToken token) throws AuthenticationException {
        String username;
        // 区分 JWT Token 和 用户名密码 Token
        if (token instanceof JwtToken) {
            // JWT 认证：从 Token 中获取用户名
            username = JwtUtil.getUsernameFromToken(((JwtToken) token).getToken());
        } else {
            // 用户名密码认证：从 UsernamePasswordToken 获取用户名
            username = (String) token.getPrincipal();
        }

        // 5. 查询数据库用户（Jeecg 自带方法）
        SysUser user = sysUserService.getUserByName(username);
        if (user == null) {
            throw new UnknownAccountException("用户不存在");
        }
        if (!user.getStatus().equals("1")) {
            throw new DisabledAccountException("用户已被禁用");
        }

        // 6. 密码加密验证（Jeecg 默认 MD5 加盐，盐值为用户名）
        ByteSource salt = ByteSource.Util.bytes(user.getUsername());
        return new SimpleAuthenticationInfo(
                user, // 存入 Subject 的用户信息
                user.getPassword(), // 数据库中的加密密码
                salt, // 盐值
                getName() // Realm 名称
        );
    }
}
```

## 四、JWT Token 封装类

```java
package com.jeecg.common.shiro;

import org.apache.shiro.authc.AuthenticationToken;

/**
 * JWT Token 封装类（替换 Shiro 原生 UsernamePasswordToken）
 */
public class JwtToken implements AuthenticationToken {

    private final String token;

    public JwtToken(String token) {
        this.token = token;
    }

    @Override
    public Object getPrincipal() {
        return token; // JWT Token 本身作为身份
    }

    @Override
    public Object getCredentials() {
        return token; // JWT Token 本身作为凭证
    }

    public String getToken() {
        return token;
    }
}
```

## 五、JWT 过滤器（全局拦截 Token）

```java
package com.jeecg.common.shiro;

import com.jeecg.common.util.JwtUtil;
import org.apache.shiro.authc.AuthenticationException;
import org.apache.shiro.authc.AuthenticationToken;
import org.apache.shiro.web.filter.authc.AuthenticatingFilter;
import org.springframework.http.HttpStatus;
import org.springframework.util.StringUtils;
import org.springframework.web.bind.annotation.RequestMethod;

import jakarta.servlet.ServletRequest;
import jakarta.servlet.ServletResponse;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import java.io.IOException;

/**
 * JWT 过滤器：拦截所有请求，验证 Token 有效性
 */
public class JwtFilter extends AuthenticatingFilter {

    private final JwtUtil jwtUtil;

    public JwtFilter(JwtUtil jwtUtil) {
        this.jwtUtil = jwtUtil;
    }

    /**
     * 创建 JWT Token
     */
    @Override
    protected AuthenticationToken createToken(ServletRequest request, ServletResponse response) {
        HttpServletRequest req = (HttpServletRequest) request;
        String token = getTokenFromRequest(req);
        if (StringUtils.hasText(token) && jwtUtil.validateToken(token)) {
            return new JwtToken(token);
        }
        return null;
    }

    /**
     * 拦截所有请求，判断是否携带 Token
     */
    @Override
    protected boolean isAccessAllowed(ServletRequest request, ServletResponse response, Object mappedValue) {
        HttpServletRequest req = (HttpServletRequest) request;
        // OPTIONS 请求直接放行（跨域预检）
        if (req.getMethod().equals(RequestMethod.OPTIONS.name())) {
            return true;
        }
        // 其他请求需要验证 Token
        return StringUtils.hasText(getTokenFromRequest(req)) && jwtUtil.validateToken(getTokenFromRequest(req));
    }

    /**
     * 无 Token 时执行此方法（返回 401 未认证）
     */
    @Override
    protected boolean onAccessDenied(ServletRequest request, ServletResponse response) throws Exception {
        HttpServletResponse resp = (HttpServletResponse) response;
        resp.setStatus(HttpStatus.UNAUTHORIZED.value());
        resp.setContentType("application/json;charset=utf-8");
        resp.getWriter().write("{\"code\":401,\"msg\":\"请先登录\"}");
        return false;
    }

    /**
     * 认证失败时执行此方法
     */
    @Override
    protected boolean onLoginFailure(AuthenticationToken token, AuthenticationException e, ServletRequest request, ServletResponse response) {
        HttpServletResponse resp = (HttpServletResponse) response;
        resp.setStatus(HttpStatus.UNAUTHORIZED.value());
        resp.setContentType("application/json;charset=utf-8");
        try {
            resp.getWriter().write("{\"code\":401,\"msg\":\"" + e.getMessage() + "\"}");
        } catch (IOException ex) {
            ex.printStackTrace();
        }
        return false;
    }

    /**
     * 从请求头获取 Token（格式：Bearer <token>）
     */
    private String getTokenFromRequest(HttpServletRequest request) {
        String header = request.getHeader("Authorization");
        if (StringUtils.hasText(header) && header.startsWith("Bearer ")) {
            return header.substring(7);
        }
        return null;
    }
}
```

## 六、Shiro 核心配置类（动态权限 + JWT 集成）

```java
package com.jeecg.common.shiro;

import com.jeecg.common.util.JwtUtil;
import com.jeecg.modules.system.service.ISysPermissionService;
import org.apache.shiro.mgt.DefaultSessionStorageEvaluator;
import org.apache.shiro.mgt.DefaultSubjectDAO;
import org.apache.shiro.spring.web.ShiroFilterFactoryBean;
import org.apache.shiro.web.mgt.DefaultWebSecurityManager;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

import jakarta.servlet.Filter;
import java.util.LinkedHashMap;
import java.util.Map;

/**
 * Jeecg Boot Shiro 核心配置（JWT + 动态权限）
 */
@Configuration
public class ShiroConfig {

    @Autowired
    private JeecgShiroRealm jeecgShiroRealm;
    @Autowired
    private JwtUtil jwtUtil;
    @Autowired
    private ISysPermissionService sysPermissionService; // Jeecg 权限服务

    /**
     * 安全管理器（无状态配置）
     */
    @Bean
    public DefaultWebSecurityManager securityManager() {
        DefaultWebSecurityManager securityManager = new DefaultWebSecurityManager();
        // 关联自定义 Realm
        securityManager.setRealm(jeecgShiroRealm);

        // 关闭 Session 存储（无状态 JWT 不需要）
        DefaultSubjectDAO subjectDAO = new DefaultSubjectDAO();
        DefaultSessionStorageEvaluator evaluator = new DefaultSessionStorageEvaluator();
        evaluator.setSessionStorageEnabled(false);
        subjectDAO.setSessionStorageEvaluator(evaluator);
        securityManager.setSubjectDAO(subjectDAO);

        return securityManager;
    }

    /**
     * Shiro 过滤器工厂（动态加载权限规则）
     */
    @Bean
    public ShiroFilterFactoryBean shiroFilterFactoryBean(DefaultWebSecurityManager securityManager) {
        ShiroFilterFactoryBean factoryBean = new ShiroFilterFactoryBean();
        factoryBean.setSecurityManager(securityManager);

        // 1. 注册 JWT 过滤器
        Map<String, Filter> filterMap = new LinkedHashMap<>();
        filterMap.put("jwt", new JwtFilter(jwtUtil));
        factoryBean.setFilters(filterMap);

        // 2. 动态加载权限规则（从数据库 sys_permission 表查询）
        Map<String, String> filterChainDefinitionMap = new LinkedHashMap<>();

        // 放行接口：登录、静态资源、Swagger
        filterChainDefinitionMap.put("/api/auth/login", "anon");
        filterChainDefinitionMap.put("/static/**", "anon");
        filterChainDefinitionMap.put("/swagger-ui/**", "anon");
        filterChainDefinitionMap.put("/v3/api-docs/**", "anon");

        // 从数据库加载动态权限（Jeecg 自带方法：查询所有权限 URL）
        // sysPermissionService.queryAllUrlPermission() 返回格式：{"/api/admin/**": "roles[ADMIN]", "/api/user/edit": "perms[user:edit]"}
        Map<String, String> dbPermissions = sysPermissionService.queryAllUrlPermission();
        filterChainDefinitionMap.putAll(dbPermissions);

        // 所有其他请求通过 JWT 过滤器
        filterChainDefinitionMap.put("/**", "jwt");

        factoryBean.setFilterChainDefinitionMap(filterChainDefinitionMap);
        return factoryBean;
    }
}
```

## 七、全局用户信息工具类

```java
package com.jeecg.common.util;

import com.jeecg.modules.system.entity.SysUser;
import org.apache.shiro.SecurityUtils;
import org.apache.shiro.subject.Subject;

/**
 * Jeecg Boot 全局用户信息工具类
 */
public class UserContextUtil {

    /**
     * 获取当前登录用户信息
     */
    public static SysUser getCurrentUser() {
        Subject subject = SecurityUtils.getSubject();
        if (subject != null && subject.getPrincipal() != null) {
            return (SysUser) subject.getPrincipal();
        }
        return null;
    }

    /**
     * 获取当前登录用户名
     */
    public static String getCurrentUsername() {
        SysUser user = getCurrentUser();
        return user != null ? user.getUsername() : null;
    }

    /**
     * 获取当前登录用户ID
     */
    public static String getCurrentUserId() {
        SysUser user = getCurrentUser();
        return user != null ? user.getId() : null;
    }
}
```

## 八、登录接口实现（生成 JWT 令牌）

```java
package com.jeecg.modules.system.controller;

import com.jeecg.common.util.JwtUtil;
import com.jeecg.common.util.UserContextUtil;
import com.jeecg.modules.system.entity.SysUser;
import org.apache.shiro.SecurityUtils;
import org.apache.shiro.authc.UsernamePasswordToken;
import org.apache.shiro.subject.Subject;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

import java.util.HashMap;
import java.util.Map;

@RestController
@RequestMapping("/api/auth")
public class AuthController {

    @Autowired
    private JwtUtil jwtUtil;

    /**
     * 登录接口：生成 JWT 令牌
     */
    @PostMapping("/login")
    public Map<String, Object> login(@RequestBody LoginRequest request) {
        Map<String, Object> result = new HashMap<>();
        try {
            Subject subject = SecurityUtils.getSubject();
            // 用户名密码认证
            UsernamePasswordToken token = new UsernamePasswordToken(request.getUsername(), request.getPassword());
            subject.login(token);

            // 获取当前用户，生成 JWT 令牌
            SysUser user = UserContextUtil.getCurrentUser();
            String jwtToken = jwtUtil.generateToken(user.getUsername());

            result.put("code", 200);
            result.put("msg", "登录成功");
            result.put("token", jwtToken);
            result.put("user", user);
        } catch (Exception e) {
            result.put("code", 500);
            result.put("msg", "登录失败：" + e.getMessage());
        }
        return result;
    }

    // 登录请求参数
    public static class LoginRequest {
        private String username;
        private String password;
        // getter/setter
        public String getUsername() { return username; }
        public void setUsername(String username) { this.username = username; }
        public String getPassword() { return password; }
        public void setPassword(String password) { this.password = password; }
    }
}
```

## 十、使用说明

1. **权限配置**：在 Jeecg Boot 系统管理后台，给角色分配菜单/权限，权限规则会自动同步到 `sys_permission` 表。
2. **接口访问**：
   - 登录：POST `/api/auth/login`，参数 `{"username":"admin","password":"123456"}`，返回 JWT 令牌。
   - 其他接口：请求头添加 `Authorization: Bearer <token>`，Shiro 会自动验证权限。
3. **全局获取用户**：在任意层调用 `UserContextUtil.getCurrentUser()` 获取当前用户信息。

## 十一、注意事项

1. 确保 `sys_permission` 表的 `url` 字段配置正确（如 `/api/admin/**`），`permission` 字段配置权限表达式（如 `roles[ADMIN]`）。
2. 生产环境需将 JWT 密钥配置在配置中心，避免硬编码。
3. 跨域问题：在 Jeecg Boot 的 `CorsConfig` 中放行 `Authorization` 请求头。
